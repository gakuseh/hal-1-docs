155extern "C" void updateDataTask(void*)
156{
157 while (!initDone) osDelay(10);
158
159 static stateestimation::AttitudeEstimator Est;
160
161 // Sustain timers for latched state transitions
162 static uint32_t ignDetectMs = 0U;
163 static uint32_t burnDetectMs = 0U;
164 static uint32_t burnoutMs = 0U; // kernel tick when burnout was confirmed
165 static bool actuation_locked = false; // permanent actuation disable latch
166
167 while (1)
168 {
169 ulTaskNotifyTake(pdTRUE, pdMS_TO_TICKS(100)); // 100 ms watchdog if sensor task stalls
170
171 if (osMutexAcquire(g_ctrls_sensor_mutex, 20) == osOK) {
172 // Apply tare requested by Initial_Condition_Tare task
173 if (g_request_tare) {
174 g_request_tare = false;
175 Est.reset(true, false); // reset attitude to identity, keep gyro bias from 3-min window
176 g_alt_pad_m = g_telemNow.altitude;
177 CTRLs_t = 0;
178
179 // Full flight state machine reset — clears any lockouts accumulated
180 // while the rocket was powered on the ground before tare.
181 STATE_IGNITION = false;
182 STATE_BURNOUT = false;
183 STATE_BACKUP_PID = true; // PID on by default until ignition
184 ALLOW_ACTUATION = false;
185 g_ctrls_enabled = false;
186 actuation_locked = false;
187 ignDetectMs = 0U;
188 burnDetectMs = 0U;
189 burnoutMs = 0U;
190 }
191
192 DataUpdate::ComputeDt();
193 g_filters.apply();
194 DataUpdate::FuseAttitude(&Est);
195 g_telemNow.verticalVelocity = DataUpdate::getVerticalVelocity();
196 // g_telemNow.altitude stays MSL; AGL offset applied at radio snapshot only
197 osMutexRelease(g_ctrls_sensor_mutex);
198 }
199
200 // Snapshot state machine inputs (benign race — worst case is one stale sample).
201 // mAccZ = remapped BMI Y axis = physical rocket vertical (reads +1g upright at rest).
202 const float az = g_telemNow.mAccZ;
203 const float pitch = g_telemNow.pitch;
204 //const float yaw = g_telemNow.yaw;
205 //const float roll = g_telemNow.roll;
206 const float vvel = g_telemNow.verticalVelocity;
207 const uint32_t nowMs = osKernelGetTickCount();
208
209 // Ignition: vertical-axis accel exceeds LIFTOFF_ACCEL_G for IGNITION_SUSTAIN_MS
210 // Latched permanently once confirmed.
211 if (!STATE_IGNITION) {
212 if (az > LIFTOFF_ACCEL_G) {
213 if (ignDetectMs == 0U) ignDetectMs = nowMs;
214 if (nowMs - ignDetectMs >= IGNITION_SUSTAIN_MS) {
215 CTRLs_t = nowMs;
216 STATE_IGNITION = true;
217 STATE_BACKUP_PID = false;
218 }
219 } else {
220 ignDetectMs = 0U;
221 }
222 }
223
224 // Burnout: vertical axis reads downward (negative) with magnitude > BURNOUT_ACCEL_G
225 // for BURNOUT_SUSTAIN_MS.
226 if (!STATE_BURNOUT) {
227 if (az < BURNOUT_ACCEL_G) {
228 if (burnDetectMs == 0U) burnDetectMs = nowMs;
229 if (nowMs - burnDetectMs >= BURNOUT_SUSTAIN_MS) {
230 STATE_BURNOUT = true;
231 burnoutMs = nowMs;
232 ALLOW_ACTUATION = !actuation_locked;
233 }
234 } else {
235 burnDetectMs = 0U;
236 }
237 }
238
239 // Permanent actuation disable: pitch tipped too far, velocity too low post-burnout,
240 // or MAX_ACTUATION_DURATION_MS has elapsed since burnout.
241 // vvel check is gated on STATE_BURNOUT — vvel is 0 on the pad and must not
242 // trip the lockout before the rocket has left the rail.
243 if (!actuation_locked) {
244 const bool timeout = STATE_BURNOUT &&
245 (nowMs - burnoutMs) >= MAX_ACTUATION_DURATION_MS;
246 const bool vvel_low = STATE_BURNOUT &&
247 fabsf(vvel) < MIN_ACTUATION_VEL_MS;
248 if (fabsf(pitch) > MAX_PITCH_ANGLE_DEG || vvel_low || timeout) {
249 actuation_locked = true;
250 ALLOW_ACTUATION = false;
251 g_ctrls_enabled = false;
252 }
253 }
254
255 // Wake CTRLs so it reads the freshest gyro data with minimum latency.
256 xTaskNotifyGive(CTRLIndicationHandle);
257 }
258}

380void CTRLs_PID()
381{
382 // Proportional-only roll-rate damper.
383 // Closed-loop: J*wdot = -(b + Keff*KP)*w => exponential decay for any KP > 0.
384 // Theoretically stable regardless of gain magnitude; larger KP = faster settling
385 // at the cost of larger canard deflections. Tune KP on the bench: spin the
386 // rocket by hand, confirm canards oppose the spin, then increase until satisfied.
387 static constexpr float KP = 0.30f; // deg deflection per deg/s roll rate -- TUNE
388 static constexpr float MAX_DEFLECT_DEG = 7.0f; // canard travel limit -- verify against stops
389 // Dead-band: suppress control below 6 deg/s to avoid fighting
390 // natural spin-up and reduce servo wear at low roll rates.
391 static constexpr float ROLL_RATE_THRESHOLD_RPS = M_PI/180.0f * 1.0; // last number is deg/sec
392
393 while (!initDone) osDelay(10);
394
395 // if (!g_ctrls_enabled) {
396 // Servos.Update(0.0f, 0.0f);
397 // continue;
398 // }
399
400 // mGyrZ = BMI.ang_vel.y = rocket spin-axis rate in deg/s (after axis remap)
401 float roll_rate_dps = 0.0f;
402 if (osMutexAcquire(g_ctrls_sensor_mutex, 2U) == osOK) {
403 roll_rate_dps = g_telemNow.mGyrZ;
404 osMutexRelease(g_ctrls_sensor_mutex);
405 }
406
407 if (fabsf(roll_rate_dps) <= ROLL_RATE_THRESHOLD_RPS) {
408 Servos.Update(0.0f, 0.0f);
409 if (osMutexAcquire(g_ctrls_sensor_mutex, 2U) == osOK) {
410 g_telemNow.servoTarget1 = g_gndData.servoOffset1;
411 g_telemNow.servoTarget2 = g_gndData.servoOffset2;
412 osMutexRelease(g_ctrls_sensor_mutex);
413 }
414 ulTaskNotifyTake(pdTRUE, pdMS_TO_TICKS(10));
415 return;
416 }
417
418 // Oppose the roll: positive rate -> negative deflection, and vice versa.
419 // "positive angle -> positive roll moment" (servo docstring), so negating
420 // the rate produces a restoring moment.
421 float cmd = -KP * roll_rate_dps;
422
423 if (cmd > MAX_DEFLECT_DEG) cmd = MAX_DEFLECT_DEG;
424 if (cmd < -MAX_DEFLECT_DEG) cmd = -MAX_DEFLECT_DEG;
425
426 // AngleOffsetDEGREES (ground-station trim) is applied inside Actuate().
427 Servos.Update(cmd, cmd);
428
429 if (osMutexAcquire(g_ctrls_sensor_mutex, 2U) == osOK) {
430 g_telemNow.servoTarget1 = cmd + g_gndData.servoOffset1;
431 g_telemNow.servoTarget2 = cmd + g_gndData.servoOffset2;
432 osMutexRelease(g_ctrls_sensor_mutex);
433 }
434
435 // Block until updateDataTask signals that a fresh sensor frame is ready.
436 // 10 ms timeout is a safety fallback only — under normal operation the
437 // notification arrives every ~2 ms (updateSensorTask rate).
438 ulTaskNotifyTake(pdTRUE, pdMS_TO_TICKS(10));
439}

470void CTRLs_STATESPACE() {
471 // ── Init: build algorithm objects ────────────────────────────────────────
472 // Mutexes (g_ctrls_sensor_mutex, g_ctrls_output_mutex) are created by
473 // ctrlsInit() in MX_FREERTOS_Init before the scheduler starts, so they are
474 // safe to acquire as soon as this task runs.
475 static RocketConfig cfg;
476 loadControlFreakRocketConfig(cfg);
477 cfg.computeDerived();
478 static Physics phys(cfg);
479 static EKF ekf(phys, cfg);
480 static ControlLaw ctrl(phys, cfg);
481
482 while (!initDone) osDelay(10);
483
484
485 // ── Update loop ──────────────────────────────────────────────────────────
486 static uint32_t prev_ms = 0U;
487 static uint32_t launch_start_ms = 0U;
488 static float model_alt_m = 0.0f;
489 static float t_zero_s = 0.0f; // subtracted from snap.flight_time_s to reset EKF clock
490 static bool apogee_latched = false;
491 static uint32_t apogee_blink_ms = 0U;
492 static bool apogee_blink_done = false;
493 static uint32_t dyn_log_ms = 0U;
494
495 float u_last_rad = 0.0f;
496 CtrlsSensorSnapshot snap = {};
497 snap.temperature_K = 288.15f;
498
499 for (;;)
500 {
501 // Remote kill or algorithm switch — return so the task loop re-evaluates.
502 if (!g_ctrls_enabled || !ALLOW_ACTUATION || STATE_BACKUP_PID) {
503 Servos.Update(0.0f, 0.0f);
504 return;
505 }
506
507 // Reset flight clock if GND commanded state-space from t=0.
508 const uint32_t now_ms = millis();
509 const float dt = (prev_ms == 0U) ? 0.01f
510 : static_cast<float>(now_ms - prev_ms) * 1e-3f;
511 prev_ms = now_ms;
512 if (dt <= 0.0f || dt > 0.5f) { osDelay(1); continue; }
513
514 // ── Read sensor snapshot ─────────────────────────────────────────────
515 bool fresh = false;
516 if (osMutexAcquire(g_ctrls_sensor_mutex, 2U) == osOK) {
517 if (g_SensorData.fresh) {
518 snap = g_SensorData;
519 g_SensorData.fresh = false;
520 fresh = true;
521 }
522 osMutexRelease(g_ctrls_sensor_mutex);
523 }
524
525 // Reset flight clock if GND commanded state-space from t=0.
526 // Capture current snap.flight_time_s as the new zero so that
527 // subsequent t values are relative to the moment of the command.
528 if (g_ctrls_reset_time) {
529 g_ctrls_reset_time = false;
530 launch_start_ms = 0U;
531 t_zero_s = snap.flight_time_s;
532 ekf.reset(cfg);
533 ctrl.reset();
534 }
535
536 const float t = snap.flight_time_s - t_zero_s;
537 const float sensor_alt_m = snap.altitude_m;
538 const float ekf_alt_for_step = fresh ? sensor_alt_m : model_alt_m;
539
540 // ── EKF step ─────────────────────────────────────────────────────────
541 if (fresh) {
542 const WeatherSample wx = phys.weatherAtAltitude(ekf_alt_for_step);
543 float va[3];
544 phys.airRelativeVelocity(ekf.xhat, wx.wind_x, wx.wind_y, va);
545 ctrl.updateRollEffectivenessSign(t, snap.gyro_rad_s[2], u_last_rad,
546 sqrtf(va[0]*va[0] + va[1]*va[1] + va[2]*va[2]));
547 MeasVec y;
548 for (int i = 0; i < 3; i++) {
549 y(i, 0) = snap.accel_g[i];
550 y(3+i, 0) = snap.gyro_rad_s[i];
551 }
552 ekf.update(t, dt, y, u_last_rad, ekf_alt_for_step);
553 } else {
554 ekf.predictOnly(t, dt, u_last_rad, ekf_alt_for_step);
555 }
556
557 // ── Sensor-aided model altitude + apogee detection ────────────────────
558 const float model_vz_ms = modelVerticalVelocityMS(ekf.xhat);
559 if (fresh)
560 model_alt_m = sensor_alt_m;
561 else
562 model_alt_m += model_vz_ms * dt;
563 if (model_alt_m < 0.0f) model_alt_m = 0.0f;
564
565 // DYN log at 10 Hz — only while controls are active
566 if (g_ctrls_enabled && !apogee_latched && now_ms - dyn_log_ms >= 20U) {
567 dyn_log_ms = now_ms;
568 const float vx = ekf.xhat(3,0), vy = ekf.xhat(4,0);
569 const float horiz_v = sqrtf(vx*vx + vy*vy);
570 const float roll_rate = ekf.xhat(2,0);
571 char dyn_ln[72];
572 snprintf(dyn_ln, sizeof(dyn_ln), "DYN,%lu\t%.2f\t%.1f\t%.2f,%.2f\t%.3f",
573 now_ms, t, model_alt_m, model_vz_ms, horiz_v, roll_rate);
574 SD_LogNewline(dyn_ln);
575 }
576
577 // Apogee: vz drops below 3 m/s after burnout
578 if (!apogee_latched && t > cfg.t_burnout && model_vz_ms < 3.0f) {
579 apogee_latched = true;
580 apogee_blink_ms = now_ms;
581 g_ctrls_enabled = false;
582 Servos.Update(0.0f, 0.0f);
583 SD_LogNewline("EVENT,ModelApogee");
584 }
585
586 // Non-blocking LED strobe for 2 s on apogee
587 if (apogee_latched && !apogee_blink_done) {
588 const uint32_t elapsed = now_ms - apogee_blink_ms;
589 if (elapsed < 2000U) {
590 HAL_GPIO_WritePin(USR_LED_GPIO_Port, USR_LED_Pin,
591 (elapsed % 200U) < 100U ? GPIO_PIN_SET : GPIO_PIN_RESET);
592 } else {
593 HAL_GPIO_WritePin(USR_LED_GPIO_Port, USR_LED_Pin, GPIO_PIN_RESET);
594 apogee_blink_done = true;
595 }
596 }
597
598 // ── Control law ───────────────────────────────────────────────────────
599 // EKF always runs above to keep the state estimate warm.
600 // Output is zeroed when controls are disabled so servos hold neutral.
601 const float control_alt = model_alt_m;
602 const float u_rad = g_ctrls_enabled
603 ? ctrl.computeControl(t, ekf.xhat, control_alt)
604 : 0.0f;
605 u_last_rad = u_rad;
606 const float u_deg = u_rad * (180.0f / static_cast<float>(M_PI));
607
608 // ── Write outputs ─────────────────────────────────────────────────────
609 if (osMutexAcquire(g_ctrls_output_mutex, 2U) == osOK) {
610 g_ctrls_canard_cmd_deg = u_deg;
611 osMutexRelease(g_ctrls_output_mutex);
612 }
613
614 // Both canards receive the same deflection command to produce roll.
615 // Ground-station offsets apply the per-servo zero-point trim.
616 Servos.Update(u_deg, u_deg);
617 if (osMutexAcquire(g_ctrls_sensor_mutex, 20) == osOK) {
618 g_telemNow.servoTarget1 = u_deg + g_gndData.servoOffset1;
619 g_telemNow.servoTarget2 = u_deg + g_gndData.servoOffset2;
620 osMutexRelease(g_ctrls_sensor_mutex);
621 }
622
623 // Block until updateDataTask signals that a fresh sensor frame is ready.
624 // 10 ms timeout is a safety fallback only.
625 ulTaskNotifyTake(pdTRUE, pdMS_TO_TICKS(10));
626 }
627}

630// CONTROLS TASK. Selects between PID and state space controllers based on flight conditions.
631// Algorithm selection is one-way: PID runs until ignition latches STATE_BACKUP_PID=false,
632// then CTRLs_STATESPACE() runs its own for(;;) loop. It returns whenever control is
633// disabled, and this loop calls it again immediately.
634// When ALLOW_ACTUATION is false, yield via ulTaskNotifyTake rather than busy-loop.
635extern "C" void CTRLs(void*)
636{
637 while (!initDone) osDelay(10);
638
639 for (;;) {
640 if (!ALLOW_ACTUATION) {
641 ulTaskNotifyTake(pdTRUE, pdMS_TO_TICKS(10));
642 continue;
643 }
644 if (STATE_BACKUP_PID) {
645 CTRLs_PID();
646 } else {
647 CTRLs_STATESPACE();
648 }
649 }

663extern "C" void Initial_Condition_Tare(void*) {
664 while (!initDone) osDelay(10);
665
666 float prevAx = 0.0f, prevAy = 0.0f, prevAz = 0.0f;
667 float prevGx = 0.0f, prevGy = 0.0f, prevGz = 0.0f;
668 uint32_t stableStart = 0U;
669 bool firstSample = true;
670
671 for (;;) {
672 osDelay(100U); // 10 Hz sampling
673
674 float ax, ay, az, gx, gy, gz;
675 if (osMutexAcquire(g_ctrls_sensor_mutex, 5) != osOK) continue;
676 ax = g_telemNow.mAccX; ay = g_telemNow.mAccY; az = g_telemNow.mAccZ;
677 gx = g_telemNow.mGyrX; gy = g_telemNow.mGyrY; gz = g_telemNow.mGyrZ;
678 osMutexRelease(g_ctrls_sensor_mutex);
679
680 if (firstSample) {
681 prevAx = ax; prevAy = ay; prevAz = az;
682 prevGx = gx; prevGy = gy; prevGz = gz;
683 stableStart = osKernelGetTickCount();
684 firstSample = false;
685 continue;
686 }
687
688 const bool stable =
689 fabsf(ax - prevAx) < TARE_ACC_DELTA_G &&
690 fabsf(ay - prevAy) < TARE_ACC_DELTA_G &&
691 fabsf(az - prevAz) < TARE_ACC_DELTA_G &&
692 fabsf(gx - prevGx) < TARE_GYRO_DELTA_DPS &&
693 fabsf(gy - prevGy) < TARE_GYRO_DELTA_DPS &&
694 fabsf(gz - prevGz) < TARE_GYRO_DELTA_DPS;
695
696 prevAx = ax; prevAy = ay; prevAz = az;
697 prevGx = gx; prevGy = gy; prevGz = gz;
698
699 if (!stable) {
700 stableStart = osKernelGetTickCount();
701 } else if (osKernelGetTickCount() - stableStart >= TARE_STABILITY_MS) {
702 g_request_tare = true;
703 vTaskSuspend(NULL); // tare requested; no further work to do
704 }
705 }
