26int32_t Magnetometer::platform_write(void* handle, uint8_t reg,
27 const uint8_t* bufp, uint16_t len)
28{
29 uint8_t tx[len + 1];
30 tx[0] = reg & 0x7F; // MSB = 0 → write
31 for (uint16_t i = 0; i < len; i++)
32 tx[i + 1] = bufp[i];
33
34 cs_low();
35 HAL_SPI_Transmit(static_cast<SPI_HandleTypeDef*>(handle), tx, len + 1, HAL_MAX_DELAY);
36 cs_high();
37 return 0;
38}

40int32_t Magnetometer::platform_read(void* handle, uint8_t reg,
41 uint8_t* bufp, uint16_t len)
42{
43 uint8_t reg_addr = reg | 0x80; // MSB = 1 → read
44
45 cs_low();
46 HAL_SPI_Transmit(static_cast<SPI_HandleTypeDef*>(handle), &reg_addr, 1, HAL_MAX_DELAY);
47 HAL_SPI_Receive (static_cast<SPI_HandleTypeDef*>(handle), bufp, len, HAL_MAX_DELAY);
48 cs_high();
49 return 0;
50}

64uint8_t Magnetometer::Init()
65{
66 uint8_t whoamI = 0;
67 lis2mdl_device_id_get(&dev_ctx, &whoamI);
68 if (whoamI != LIS2MDL_ID)
69 return 1;
70
71 lis2mdl_reset_set(&dev_ctx, PROPERTY_ENABLE);
72 uint8_t rst;
73 do {
74 lis2mdl_reset_get(&dev_ctx, &rst);
75 } while (rst);
76
77 lis2mdl_block_data_update_set(&dev_ctx, PROPERTY_ENABLE);
78 lis2mdl_data_rate_set(&dev_ctx, LIS2MDL_ODR_100Hz);
79 lis2mdl_offset_temp_comp_set(&dev_ctx, PROPERTY_ENABLE);
80 lis2mdl_operating_mode_set(&dev_ctx, LIS2MDL_CONTINUOUS_MODE);
81
82 _sensor_active = true;
83 return 0;
84}

86// ─── Update ───────────────────────────────────────────────────────────────────
87
88uint8_t Magnetometer::Update()
89{
90 if (!_sensor_active) return 1;
91
92 uint8_t drdy = 0;
93 if (lis2mdl_mag_data_ready_get(&dev_ctx, &drdy) != 0)
94 return 2;
95
96 if (!drdy) return 0; // no new sample — retain last valid _raw
97
98 int16_t mag_raw[3] = {};
99 if (lis2mdl_magnetic_raw_get(&dev_ctx, mag_raw) != 0)
100 return 3;
101
102 int16_t temp_raw = 0;
103 if (lis2mdl_temperature_raw_get(&dev_ctx, &temp_raw) != 0)
104 return 4;
105
106 // lis2mdl_from_lsb_to_mgauss returns milli-Gauss; convert to Gauss so
107 // the attitude estimator receives "preferably Gauss" units directly.
108 _raw.B_x = lis2mdl_from_lsb_to_mgauss(mag_raw[0]) * 0.001f;
109 _raw.B_y = lis2mdl_from_lsb_to_mgauss(mag_raw[1]) * 0.001f;
110 _raw.B_z = lis2mdl_from_lsb_to_mgauss(mag_raw[2]) * 0.001f;
111 _raw.LIS2_Temperature = lis2mdl_from_lsb_to_celsius(temp_raw);
112
113 return 0;
114}
