22 // ── Altitude / pressure ───────────────────────────────────────────────────
23 g_telemNow.altitude = pressureFilter.update(g_telemNow.altitude);
24 g_SensorData.altitude_m = g_telemNow.altitude;
25
26 // ── Acceleration (g units, Y/Z axis swap already applied by Sensors::Update) ─
27 accelFilter.update(g_telemNow.mAccX, g_telemNow.mAccY, g_telemNow.mAccZ);
28 g_SensorData.accel_g[0] = g_telemNow.mAccX;
29 g_SensorData.accel_g[1] = g_telemNow.mAccY;
30 g_SensorData.accel_g[2] = g_telemNow.mAccZ;
31
32 // ── Gyroscope (deg/s in g_telemNow; EKF snapshot expects rad/s) ───────────
33 gyroFilter.update(g_telemNow.mGyrX, g_telemNow.mGyrY, g_telemNow.mGyrZ);
34 g_SensorData.gyro_rad_s[0] = g_telemNow.mGyrX * DEG_TO_RAD;
35 g_SensorData.gyro_rad_s[1] = g_telemNow.mGyrY * DEG_TO_RAD;
36 g_SensorData.gyro_rad_s[2] = g_telemNow.mGyrZ * DEG_TO_RAD;
37}
