14void DataUpdate::ComputeDt()
15{
16 // Use raw DWT cycle counts rather than micros().
17 //
18 // micros() = DWT->CYCCNT / us_f, which wraps at (2^32 / us_f) ≈ 8.95 s
19 // at 480 MHz — not at 2^32. Subtracting two values that wrap at a
20 // non-power-of-two boundary with uint32 arithmetic gives a spurious
21 // ~4286 s spike every ~8.95 s, which caused the Madgwick integrator to
22 // apply a massive rotation step and produce the periodic quaternion jumps.
23 //
24 // DWT->CYCCNT wraps at exactly 2^32 cycles, so uint32 subtraction is
25 // always correct. We convert to seconds only after the subtraction.
26 static uint32_t lastCycles = 0;
27 const uint32_t now = DWT->CYCCNT;
28 if (lastCycles == 0) {
29 g_dt = 0.0f;
30 } else {
31 const uint32_t elapsed = now - lastCycles; // correct across 2^32 rollover
32 g_dt = static_cast<float>(elapsed) /
33 static_cast<float>(HAL_RCC_GetHCLKFreq());
34 }
35 lastCycles = now;
36}

38void DataUpdate::FuseAttitude(stateestimation::AttitudeEstimator* est)
39{
40 if (g_dt <= 0.0f) return;
41
42 static constexpr double DEG_TO_RAD = 3.14159265358979323846 / 180.0;
43 static constexpr double RAD_TO_DEG = 180.0 / 3.14159265358979323846;
44
45 // FlightComputer_SENSORS already remapped BMI axes so that:
46 // mAccX / mGyrX = BMI.x (lateral 1)
47 // mAccY / mGyrY = BMI.z (lateral 2)
48 // mAccZ / mGyrZ = BMI.y (rocket longitudinal axis, pointing up)
49 // Pass these directly: no second remap. The estimator's Z axis = rocket-up.
50 const double gx = g_telemNow.mGyrX * DEG_TO_RAD;
51 const double gy = g_telemNow.mGyrY * DEG_TO_RAD;
52 const double gz = g_telemNow.mGyrZ * DEG_TO_RAD;
53 const double ax = g_telemNow.mAccX;
54 const double ay = g_telemNow.mAccY;
55 const double az = g_telemNow.mAccZ;
56
57 est->update(static_cast<double>(g_dt),
58 gx, gy, gz,
59 ax, ay, az,
60 0.0, 0.0, 0.0); // no magnetometer
61
62 double q[4];
63 est->getAttitude(q); // q[0]=w, q[1]=x, q[2]=y, q[3]=z
64 g_telemNow.Qw = static_cast<float>(q[0]);
65 g_telemNow.Qx = static_cast<float>(q[1]);
66 g_telemNow.Qy = static_cast<float>(q[2]);
67 g_telemNow.Qz = static_cast<float>(q[3]);
68
69 // ZYX Euler convention with Z = rocket longitudinal axis:
70 // eulerYaw() = rotation about Z = spin about rocket axis → physical roll
71 // eulerPitch() = rotation about Y = lateral tilt → physical yaw
72 // eulerRoll() = rotation about X = lateral tilt → physical pitch
73 g_telemNow.roll = static_cast<float>(est->eulerPitch() * RAD_TO_DEG);
74 g_telemNow.yaw = static_cast<float>(est->eulerYaw() * RAD_TO_DEG);
75 g_telemNow.pitch = static_cast<float>(est->eulerRoll() * RAD_TO_DEG);
76}

78float DataUpdate::getVerticalVelocity()
79{
80 static uint32_t lastVvelMs = 0U;
81 static float lastAlt = 0.0f;
82 static float lastVvel = 0.0f;
83
84 const uint32_t nowMs = HAL_GetTick();
85 const float altNow = g_telemNow.altitude;
86
87 if (lastVvelMs == 0U) {
88 lastVvelMs = nowMs;
89 lastAlt = altNow;
90 return 0.0f;
91 }
92
93 const uint32_t elapsed = nowMs - lastVvelMs;
94 if (elapsed < VVEL_UPDATE_PERIOD_MS) {
95 return lastVvel;
96 }
97
98 if (fabs(altNow - lastAlt) < 0.1f) { // delta v < 0.1f = m/s
99 lastVvel = 0.0f;
100 lastVvelMs = nowMs;
101 lastAlt = altNow;
102 return 0.0f;
103 }
104
105 float vvel = (altNow - lastAlt) / (static_cast<float>(elapsed) * 1.0e-3f);
106 // Clamp to physically plausible range — suppresses ADC glitches and
107 // timing anomalies from producing absurd readings that could trip the
108 // actuation lockout or ignition state machine.
109 if (vvel > 300.0f) vvel = 300.0f;
110 if (vvel < -300.0f) vvel = -300.0f;
111
112 lastVvel = vvel;
113 lastVvelMs = nowMs;
114 lastAlt = altNow;
115 return lastVvel;
116}
