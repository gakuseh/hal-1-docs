24int32_t ISM_SPI_Write(void *handle,
25 uint8_t reg,
26 const uint8_t *bufp,
27 uint16_t len) {
28 SPI_HandleTypeDef *hspi = (SPI_HandleTypeDef*)handle;
29
30 ISM_GTX[0] = reg & 0x7F; // Write
31 if (len > 1)
32 ISM_GTX[0] |= 0x40;
33
34 memcpy(&ISM_GTX[1], bufp, len);// Auto-increment
35
36 ISM_CS_Low();
37 HAL_SPI_Transmit(hspi, ISM_GTX, len+1, 5000);
38 while(hspi->State == HAL_SPI_STATE_BUSY) {}
39 ISM_CS_High();
40
41 return 0;
42}

44int32_t ISM_SPI_Read(void *handle,
45 uint8_t reg,
46 uint8_t *bufp,
47 uint16_t len) {
48 auto *hspi = static_cast<SPI_HandleTypeDef *>(handle);
49
50 ISM_GTX[0] = reg | 0x80; // Read
51 if (len > 1)
52 ISM_GTX[0] |= 0x40; // Auto-increment
53
54 ISM_CS_Low();
55 HAL_SPI_TransmitReceive(hspi, ISM_GTX, ISM_GRX, len+1, 5000);
56 while(hspi->State == HAL_SPI_STATE_BUSY) {}
57 ISM_CS_High();
58
59 memcpy(bufp, ISM_GRX + 1, len);
60 return 0;
61}
