115int TEST::SD_TEST()
116{
117 // Phase 1 — Init
118 int8_t res = SD_Init();
119 if (res != 0) {
120 // Fast blink: init failed
121 while (1) {
122 HAL_GPIO_TogglePin(USR_LED_GPIO_Port, USR_LED_Pin);
123 vTaskDelay(pdMS_TO_TICKS(100));
124 }
125 }
126
127 uint32_t count = 0;
128
129 // Phase 2 — Write loop (10 iterations)
130 while (count < 10) {
131
132 // Write header on first pass only
133 if (count == 0) {
134 res = SD_LogNewline("=== SD TEST START ===");
135 if (res != 0) {
136 while (1) {
137 HAL_GPIO_TogglePin(USR_LED_GPIO_Port, USR_LED_Pin);
138 vTaskDelay(pdMS_TO_TICKS(100));
139 }
140 }
141 }
142
143 // Write a GPS row with fake but distinct values
144 res = SD_LogGPS(
145 43.4723000 + count * 0.0000001, // lat
146 -80.5449000 + count * 0.0000001, // lon
147 330.00 + count * 0.5, // alt
148 12, // hour
149 0, // min
150 (uint8_t)(count * 6) // sec (0, 6, 12, ... 54)
151 );
152 if (res != 0) {
153 while (1) {
154 HAL_GPIO_TogglePin(USR_LED_GPIO_Port, USR_LED_Pin);
155 vTaskDelay(pdMS_TO_TICKS(100));
156 }
157 }
158
159 // Write inline counter string to verify inline append
160 char buf[32];
161 snprintf(buf, sizeof(buf), "row:%lu", count);
162 res = SD_LogInline(buf);
163 if (res != 0) {
164 while (1) {
165 HAL_GPIO_TogglePin(USR_LED_GPIO_Port, USR_LED_Pin);
166 vTaskDelay(pdMS_TO_TICKS(100));
167 }
168 }
169
170 HAL_GPIO_TogglePin(USR_LED_GPIO_Port, USR_LED_Pin);
171 vTaskDelay(pdMS_TO_TICKS(1000));
172 count++;
173 }
174
175 // Phase 3 — Clean close
176 SD_LogNewline("=== SD TEST END ===");
177 SD_Close();
178
179 // Slow blink: test completed cleanly, safe to remove card
180 while (1) {
181 HAL_GPIO_TogglePin(USR_LED_GPIO_Port, USR_LED_Pin);
182 vTaskDelay(pdMS_TO_TICKS(500));
183 }
184
185 return 0;
186}
