20int8_t SD_Init()
21{
22 if (s_open) return 0;
23
24 if (!s_mutex) s_mutex = osMutexNew(&k_sd_mutex_attr);
25
26 // Guard: verify FATFS driver was linked by MX_FATFS_Init (retUSER == 0).
27 // Returns -20 (not a valid FRESULT) so it is distinguishable in the debugger.
28 extern uint8_t retUSER;
29 if (retUSER != 0) return -20;
30
31 // f_mount failure: return -(FRESULT). FR_NOT_READY=-3, FR_NO_FILESYSTEM=-13, etc.
32 // Check sd_init_fail_step (sd_diskio.c) to see exactly where disk_initialize stopped.
33 FRESULT res = f_mount(&USERFatFS, USERPath, 1);
34 if (res != FR_OK) return -(int8_t)res;
35
36 // f_open failure: offset FRESULT by -100 so mount vs open errors are distinguishable.
37 res = f_open(&s_file, "HAL1_LOG.TXT", FA_OPEN_ALWAYS | FA_WRITE);
38 if (res != FR_OK) return (int8_t)(-100 - (int8_t)res);
39
40 f_lseek(&s_file, f_size(&s_file)); // seek to end for append behaviour
41 s_open = true;
42 return 0;
43}

45int8_t SD_LogNewline(const char* msg)
46{
47 if (!s_open || !s_mutex) return -1;
48 if (osMutexAcquire(s_mutex, pdMS_TO_TICKS(50)) != osOK) return -4;
49 int8_t ret = 0;
50 if (f_printf(&s_file, "%s\n", msg) < 0) ret = -2;
51 else if (f_sync(&s_file) != FR_OK) ret = -3;
52 osMutexRelease(s_mutex);
53 return ret;
54}

67int8_t SD_LogGPS(double lat, double lon, double alt,
68 uint8_t hour, uint8_t min, uint8_t sec)
69{
70 if (!s_open || !s_mutex) return -1;
71 char buf[64];
72 snprintf(buf, sizeof(buf), "%.7f,%.7f,%.2f,%02u,%02u,%02u\n",
73 lat, lon, alt, (unsigned)hour, (unsigned)min, (unsigned)sec);
74 if (osMutexAcquire(s_mutex, pdMS_TO_TICKS(50)) != osOK) return -4;
75 int8_t ret = 0;
76 if (f_puts(buf, &s_file) == EOF) ret = -2;
77 else if (f_sync(&s_file) != FR_OK) ret = -3;
78 osMutexRelease(s_mutex);
79 return ret;

88{
