24Baro_Unified::BMP_Status Baro_Unified::Update() {
25 status.A = SensorA.Update();
26 data.SensorA = SensorA.getRawData();
27
28 if (TMR_on) {
29 status.B = SensorB.Update();
30 status.C = SensorC.Update();
31 data.SensorB = SensorB.getRawData();
32 data.SensorC = SensorC.getRawData();
33
34 data.Filtered.Pressure = TMRFloat::Vote(data.SensorA.Pressure, data.SensorB.Pressure, data.SensorC.Pressure);
35 data.Filtered.heightMeters = TMRFloat::Vote(data.SensorA.heightMeters, data.SensorB.heightMeters, data.SensorC.heightMeters);
36 data.Filtered.Temperature = TMRFloat::Vote(data.SensorA.Temperature, data.SensorB.Temperature, data.SensorC.Temperature);
37 } else {
38 // single-sensor mode: Filtered mirrors primary sensor
39 data.Filtered = data.SensorA;
40 }
41 return status;
42}
