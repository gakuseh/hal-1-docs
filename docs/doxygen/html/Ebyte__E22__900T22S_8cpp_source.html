16/* Mutex for thread-safe radio access */
17
18// Because we have multiple things that can't overlap,
19// we can use FreeRTOS's mutex to handle everything.
20// using the pair xSemaphoreTake() and xSemaphoreGive(),
21// whatever code we put in between will not be run unless
22// no other code is queued. This is useful because we
23// don't want things like commands and data to overlap.
24// If they did, it would cause bytes to be corrupted and
25// give bad data. Here, the semaphore stuff prevents
26// cases like config commands being sent, then in the
27// middle of its execution, starting a data transfer.

71void changeMode(EBYTE_MODE mode)
72{
73 // Module will finish its current task (e.g. transmission) even if
74 // the mode is switched. No delay is needed
75 switch(mode)
76 {
77 case TRANS:
78 HAL_GPIO_WritePin(e22_cfg.E22_M0_PORT, e22_cfg.E22_M0_PIN, GPIO_PIN_RESET);
79 HAL_GPIO_WritePin(e22_cfg.E22_M1_PORT, e22_cfg.E22_M1_PIN, GPIO_PIN_RESET);
80 break;
81
82 case WOR:
83 HAL_GPIO_WritePin(e22_cfg.E22_M0_PORT, e22_cfg.E22_M0_PIN, GPIO_PIN_SET);
84 HAL_GPIO_WritePin(e22_cfg.E22_M1_PORT, e22_cfg.E22_M1_PIN, GPIO_PIN_RESET);
85 break;
86
87 case CONFIG:
88 HAL_GPIO_WritePin(e22_cfg.E22_M0_PORT, e22_cfg.E22_M0_PIN, GPIO_PIN_RESET);
89 HAL_GPIO_WritePin(e22_cfg.E22_M1_PORT, e22_cfg.E22_M1_PIN, GPIO_PIN_SET);
90 break;
91
92 case OFF:
93 HAL_GPIO_WritePin(e22_cfg.E22_M0_PORT, e22_cfg.E22_M0_PIN, GPIO_PIN_SET);
94 HAL_GPIO_WritePin(e22_cfg.E22_M1_PORT, e22_cfg.E22_M1_PIN, GPIO_PIN_SET);
95 break;
96 }
97
98 // Wait until the module has processed the mode switch
99 waitAux_e22_900t22s(200);
100}

105{
106 if (cfg == NULL)
107 return E22_ERR_INVALID_PARAM;
108
109 /* copy desired configuration locally */
110 e22_cfg = *cfg;
111
112 /* CONFIG mode on the E22 always uses fixed 9600 baud regardless of REG0.
113 * Force the MCU UART to 9600 now so config read/write succeeds even if
114 * CubeMX regenerated usart.c at a different rate, or a prior init already
115 * bumped it to 115200. It will be raised to the target rate at the end. */
116 e22_cfg.huart->Init.BaudRate = 9600;
117 HAL_UART_Init(e22_cfg.huart);
118
119 /* ensure mutex exists */
120 // if (e22_mutex == NULL)
121 // e22_mutex = xSemaphoreCreateMutex();
122 //
123 // if (e22_mutex == NULL)
124 // return E22_ERR_BUSY;
125
126 // active-low so this turns the radio on
127 HAL_GPIO_WritePin(RADIO_RST_GPIO_Port, RADIO_RST_Pin, GPIO_PIN_SET);
128
129 /* ensure radio is ready */
130 waitAux_e22_900t22s(5000);
131
132 // for safety
133 vTaskDelay(pdMS_TO_TICKS(500));
134 waitAux_e22_900t22s(1000);
135
136 /* read current radio configuration */
137 config_e22_900t22s current_cfg = {};
138
139 if (writeConfig_e22_900t22s(cfg, true) != E22_OK)
140 return E22_ERR_UART;
141
142 if (readConfig_e22_900t22s(&current_cfg) != E22_OK)
143 return E22_ERR_UART;
144
145 /* compare relevant fields */
146 bool config_matches =
147 (current_cfg.ADDH == cfg->ADDH) &&
148 (current_cfg.ADDL == cfg->ADDL) &&
149 (current_cfg.NETID == cfg->NETID) &&
150 (current_cfg.REG0 == cfg->REG0) &&
151 (current_cfg.REG1 == cfg->REG1) &&
152 (current_cfg.REG2 == cfg->REG2) &&
153 (current_cfg.REG3 == cfg->REG3) ;
154

158 if (writeConfig_e22_900t22s(cfg, true) != E22_OK)
159 return E22_ERR_UART;
160
161 if (waitAux_e22_900t22s(1000) != E22_OK)
162 return E22_ERR_TIMEOUT;
163 }
164
165 if (readConfig_e22_900t22s(&current_cfg) != E22_OK)
166 return E22_ERR_UART;
167 config_matches =
168 (current_cfg.ADDH == cfg->ADDH) &&
169 (current_cfg.ADDL == cfg->ADDL) &&
170 (current_cfg.NETID == cfg->NETID) &&
171 (current_cfg.REG0 == cfg->REG0) &&
172 (current_cfg.REG1 == cfg->REG1) &&
173 (current_cfg.REG2 == cfg->REG2) &&
174 (current_cfg.REG3 == cfg->REG3);

232// save_to_flash overwrites the factory default settings.
233// Set to TRUE when deploying finalized code!
234int8_t writeConfig_e22_900t22s(
235 const config_e22_900t22s *cfg,
236 bool save_to_flash)
237{
238 uint8_t frame[10];
239 uint8_t _echo[10] = {0};
240
241 // Head command byte
242 frame[0] = save_to_flash ?
243 COMMAND_BYTE_WRITE_CFG_SAVE_FLASH :
244 COMMAND_BYTE_WRITE_CFG_NOSAVE_FLASH;
245 frame[1] = 0x00;
246 frame[2] = 0x07;
247
248 // REG3 is for more advanced functions and is not included
249 frame[3] = cfg->ADDH;
250 frame[4] = cfg->ADDL;
251 frame[5] = cfg->NETID;
252 frame[6] = cfg->REG0;
253 frame[7] = cfg->REG1;
254 frame[8] = cfg->REG2;
255 frame[9] = cfg->REG3;
256
257 //xSemaphoreTake(e22_mutex, portMAX_DELAY);
258
259
260 if (HAL_GPIO_ReadPin(M0_Radio_GPIO_Port, M0_Radio_Pin) != GPIO_PIN_RESET ||
261 HAL_GPIO_ReadPin(M1_Radio_GPIO_Port, M1_Radio_Pin) != GPIO_PIN_SET)
262 changeMode(CONFIG);
263 // Ensure that config can be written
264 vTaskDelay(pdMS_TO_TICKS(500));
265 waitAux_e22_900t22s(1000);
266
267 // Construct and write config commands
268
269 // Flush any stale bytes (e.g. from a mode-switch glitch when re-entering
270 // CONFIG while already in CONFIG, or leftovers from a prior operation).
271 while (__HAL_UART_GET_FLAG(&huart8, UART_FLAG_RXFNE))
272 (void)huart8.Instance->RDR;
273
274 if(uartWrite(frame, 10) != E22_OK)
275 {
276 //xSemaphoreGive(e22_mutex);
277 return E22_ERR_UART;
278 }
279
280 /* The E22 echoes back the written registers as confirmation:
281 * [C1 00 07 ADDH ADDL NETID REG0 REG1 REG2 REG3] (10 bytes).
282 * Must be consumed here — it sits in the MCU UART buffer regardless of
283 * when changeMode(TRANS) is called, and shifts the next uartRead() by 1 byte. */
284 if (uartRead(_echo, 10) != E22_OK)
285 return E22_ERR_UART;
286
287 if (_echo[0] != COMMAND_BYTE_READ_CFG || _echo[1] != 0x00 || _echo[2] != 0x07)
288 return E22_ERR_UART;
289
290 changeMode(TRANS);
291
292 e22_cfg = *cfg;
293 //xSemaphoreGive(e22_mutex);
294 return E22_OK;
295}

297int8_t readConfig_e22_900t22s(config_e22_900t22s *cfg)
298{
299 uint8_t cmd[3];
300 cmd[0] = COMMAND_BYTE_READ_CFG;
301 cmd[1] = 0x00; // Start from ADDH
302 cmd[2] = 0x07; // Read all necessary registers
303
304 //xSemaphoreTake(e22_mutex, portMAX_DELAY);
305
306 if (HAL_GPIO_ReadPin(M0_Radio_GPIO_Port, M0_Radio_Pin) != GPIO_PIN_RESET ||
307 HAL_GPIO_ReadPin(M1_Radio_GPIO_Port, M1_Radio_Pin) != GPIO_PIN_SET)
308 changeMode(CONFIG);
309
310 // Process mode switch in case aux pin logic is messed
311 vTaskDelay(pdMS_TO_TICKS(400));
312 waitAux_e22_900t22s(1000); // shouldn't need this but just in case
313
314 if (uartWrite(cmd, 3) != E22_OK)
315 return E22_ERR_UART;
316
317 uint8_t resp[10] = {};
318
319 int8_t rslt = uartRead(resp,10);
320 if(rslt != E22_OK)
321 {
322 //xSemaphoreGive(e22_mutex);
323 return E22_ERR_UART;
324 }
325
326 // first three values are just repetition of sent data
327 if (resp[0] != cmd[0] ||
328 resp[1] != cmd[1] ||
329 resp[2] != cmd[2])
330 return E22_ERR_DATA_VERIFICATION;
331
332 cfg->ADDH = resp[3];
333 cfg->ADDL = resp[4];
334 cfg->NETID = resp[5];
335 cfg->REG0 = resp[6];
336 cfg->REG1 = resp[7];
337 cfg->REG2 = resp[8];
338 cfg->REG3 = resp[9];
339
340 //xSemaphoreGive(e22_mutex);
341
342 changeMode(TRANS);
343
344 return E22_OK;
345}

406// ---------------------------------------------------------------------------
407// Returns true if the E22 is currently clocking a received packet out onto
408// UART (AUX low) OR if at least one byte has already landed in the UART RX

422int8_t get_rssi_e22_900t22s(void)
423{
424 // only callable in trans or WOR mode
425 if ((HAL_GPIO_ReadPin(M0_Radio_GPIO_Port, M0_Radio_Pin) == GPIO_PIN_RESET &&
426 HAL_GPIO_ReadPin(M1_Radio_GPIO_Port, M1_Radio_Pin) == GPIO_PIN_RESET) ||
427 (HAL_GPIO_ReadPin(M0_Radio_GPIO_Port, M0_Radio_Pin) == GPIO_PIN_SET &&
428 HAL_GPIO_ReadPin(M1_Radio_GPIO_Port, M1_Radio_Pin) == GPIO_PIN_RESET))
429 {
430 static uint8_t RSSISend[6] = {0xC0, 0xC1, 0xC2, 0xC3, 0x00, 0x02};
431 uint8_t RSSIReceive[5] = {};
432
433 // Flush any stale bytes (preamble zeros, prior echo leftovers) before read
434 __HAL_UART_FLUSH_DRREGISTER(e22_cfg.huart);
435
436 waitAux_e22_900t22s(200);
437 uartWrite(RSSISend, 6);
438 uartRead(RSSIReceive, 5);
439
440 // all bytes OK
441 for (int i=0; i<2; i++) {
442 if (RSSIReceive[i] == 0xC1 && RSSIReceive[i+1] == 0x00 && RSSIReceive[i+2] == 0x02)
443 {
444 // byte 3 is the current RSSI which doesn't have purpose for the FC. We want
445 // the RSSI of the signal
446 last_rssi = RSSIReceive[4];
447 return static_cast<int8_t>(last_rssi - 256);
448 }
449 }
450
451 return 0;
452 }
453
454 // Wrong mode
455 return -1;
456}

458/*
459 * Single-phase receive sized to the exact expected packet.
460 *
461 * Reads exactly (5 + expected_payload_len + 4) bytes — the full frame:
462 * [SYNC1][SYNC2][len][seq_lo][seq_hi][payload x len][crc_lo][crc_hi][SYNC2][SYNC1]
463 *
464 * Timeout is derived from the actual byte count and UART baud rate so it
465 * scales automatically with payload size and stays as tight as possible.
466 * A 25 ms margin covers the E22's air-to-UART pipeline flush.
467 * The HAL_UART_Receive call is kept as a single blocking transfer so the
468 * receive is never interrupted mid-packet.
469 *
470 * SYNC validation and length/CRC checking are left to decodeData() in
471 * Telemetry.cpp — this function only guarantees a full buffer fill.
472 *
473 * Returns total bytes placed in buffer or < 0 on error:
474 * -1 HAL_ERROR or HAL_BUSY
475 * -2 timeout with fewer than 7 bytes received (unusable)
476 */
477int16_t recieve_e22_900t22s(uint8_t *buffer, uint16_t expected_payload_len)
478{
479 // pkt_total: the framed packet bytes [SYNC1..SYNC1].
480 // total: +1 for the hardware RSSI byte the E22 appends when R3_7_RSSI_BYTE_ENABLE is set.
481 // +4 guard bytes: absorb up to 4 stale bytes that may be left in the UART FIFO
482 // from a previous timed-out call (typically the orphaned RSSI byte = 0xD9).
483 // decodeData() scans the whole buffer for SYNC1/SYNC2, so the offset is harmless.
484 const uint16_t pkt_total = 5u + expected_payload_len + 4u;
485 const uint16_t total = pkt_total + 1u + 4u;
486
487 // Timeout derived from byte count + 50 ms wireless pipeline margin.
488 const uint32_t timeout_ms = ((uint32_t)total * 10u * 1000u) / 38400u + 50u;
489
490 HAL_StatusTypeDef status = HAL_UART_Receive(&huart8, buffer, total, timeout_ms);
491
492 if (status == HAL_OK)
493 return (int16_t)total;
494
495 uint16_t received = total - huart8.RxXferCount;
496 if (received < 7u)
497 return -2; // not enough bytes for any valid packet
498 return (int16_t)received;
499}
