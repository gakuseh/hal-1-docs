54int8_t Sensors::Update()
55{
56 IMU.Update();
57 Baro.Update();
58 GPS.update();
59
60 // Servos' positions are updated in control algo (global Servos in main.cpp).
61 // SetOffset is NOT called here — calling Actuate() from this task at 500 Hz
62 // would overwrite TIM3 CCR1/CCR2 and fight the CTRLs commands.
63 SAsym<float> c_Servos = Servos.getCurrentAngle();
64 MAXM10S::gpsData c_GPS = GPS.getData();
65 g_BMI = IMU.getVotedBMI();
66 g_BMP = Baro.getData().Filtered;
67
68 static constexpr float DEG_TO_RAD = 3.14159265358979323846f / 180.0f;
69 static uint32_t liftoffDetectMs = 0U;
70 static uint32_t liftoffMs = 0U;
71
72 const uint32_t nowMs = HAL_GetTick();
73 if (liftoffMs == 0U) {
74 if (g_BMI.accel_linear.magnitude() > LIFTOFF_ACCEL_G) {
75 if (liftoffDetectMs == 0U) liftoffDetectMs = nowMs;
76 if (static_cast<uint32_t>(nowMs - liftoffDetectMs) > LIFTOFF_SUSTAIN_MS) {
77 liftoffMs = liftoffDetectMs;
78 }
79 } else {
80 liftoffDetectMs = 0U;
81 }
82 }
83
84 const float flightTimeS =
85 (liftoffMs == 0U) ? 0.0f : static_cast<float>(nowMs - liftoffMs) * 1.0e-3f;
86
87 // Sensors mutex is acquired when updating g_SensorData and g_telemNow.
88 // The return value MUST be checked: releasing a mutex you don't own corrupts
89 // FreeRTOS internal state and triggers configASSERT.
90 if (osMutexAcquire(g_ctrls_sensor_mutex, 20) == osOK) {
91
92 g_telemNow.latitude = c_GPS.latitude;
93 g_telemNow.longitude = c_GPS.longitude;
94 g_telemNow.GPSaltitude = c_GPS.altitude;
95 g_GPS.latitude = c_GPS.latitude;
96 g_GPS.longitude = c_GPS.longitude;
97 g_GPS.altitude = c_GPS.altitude;
98 g_GPS.hour = c_GPS.hour;
99 g_GPS.minute = c_GPS.minute;
100 g_GPS.second = c_GPS.second;
101 g_GPS.satellites = c_GPS.satellites;
102 g_GPS.hdop = c_GPS.hdop;
103 g_telemNow.altitude = g_BMP.heightMeters; // absolute height — delta computed in getVerticalVelocity
104 g_telemNow.temperature = g_BMP.Temperature;
105
106 // NOTE THAT Y AND Z ARE FLIPPED
107 // Y points upwards in the IMU, but the controls algorithm expects that
108 // Z is upwards.
109 g_telemNow.mAccX = g_BMI.accel_linear.x;
110 g_telemNow.mAccY = g_BMI.accel_linear.z;
111 g_telemNow.mAccZ = g_BMI.accel_linear.y;
112 g_telemNow.mGyrX = g_BMI.ang_vel.x;
113 g_telemNow.mGyrY = g_BMI.ang_vel.z;
114 g_telemNow.mGyrZ = g_BMI.ang_vel.y;
115 g_telemNow.servoPos1 = c_Servos.S1;
116 g_telemNow.servoPos2 = c_Servos.S2;
117
118 // Populate controls sensor snapshot (same axes/units as g_telemNow above).
119 // ang_vel from BMI is in deg/s — EKF expects rad/s.
120 g_SensorData.accel_g[0] = g_BMI.accel_linear.x;
121 g_SensorData.accel_g[1] = g_BMI.accel_linear.z;
122 g_SensorData.accel_g[2] = g_BMI.accel_linear.y;
123 g_SensorData.gyro_rad_s[0] = g_BMI.ang_vel.x * DEG_TO_RAD;
124 g_SensorData.gyro_rad_s[1] = g_BMI.ang_vel.z * DEG_TO_RAD;
125 g_SensorData.gyro_rad_s[2] = g_BMI.ang_vel.y * DEG_TO_RAD;
126 g_SensorData.altitude_m = g_BMP.heightMeters;
127 // Kept for telemetry/diagnostics; controls use weather-profile temperature.
128 g_SensorData.temperature_K = g_BMP.Temperature + 273.15f;
129 g_SensorData.flight_time_s = flightTimeS;
130 g_SensorData.timestamp_ms = HAL_GetTick();
131 g_SensorData.fresh = true;
132
133 osMutexRelease(g_ctrls_sensor_mutex);
134 } // osMutexAcquire
135
136 return STATUS_OK;
137}
