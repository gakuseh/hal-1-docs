15int SIMU_BMI323::Init(BMI_INDEX DeviceNum)
16{
17 _obj.index = DeviceNum;
18 _obj.SPIbus = &hspi4;
19
20 /* Map sensor index to chip select pins */
21 switch(_obj.index)
22 {
23 case SENSOR1_I:
24 _obj.cs_port = SIMU_CS1_GPIO_Port;
25 _obj.cs_pin = SIMU_CS1_Pin;
26 break;
27
28 case SENSOR2_I:
29 _obj.cs_port = SIMU_CS2_GPIO_Port;
30 _obj.cs_pin = SIMU_CS2_Pin;
31 break;
32
33 case SENSOR3_I:
34 _obj.cs_port = SIMU_CS3_GPIO_Port;
35 _obj.cs_pin = SIMU_CS3_Pin;
36 break;
37
38 default:
39 return 1;
40 }
41
42 device.intf = BMI3_SPI_INTF;
43 device.intf_ptr = &_obj;
44
45 device.read = BMI3_SPI_Read;
46 device.write = BMI3_SPI_Write;
47 device.delay_us = BMI323_delay_us;
48
49 if(bmi323_init(&device) != BMI3_OK)
50 {
51 return 2;
52 }
53
54 config[0].type = BMI323_ACCEL;
55 config[1].type = BMI323_GYRO;
56
57 if(bmi323_get_sensor_config(config, 2, &device) != BMI3_OK)
58 return 3;
59
60 map_int.acc_drdy_int = BMI3_INT1;
61 map_int.gyr_drdy_int = BMI3_INT1;
62 map_int.temp_drdy_int = BMI3_INT1;
63
64 if (bmi323_map_interrupt(map_int, &device) != BMI3_OK) {
65 return 4;
66 }
67
68 // 200 Hz ODR: matches the CTRLs task rate and gives fresh data every 5 ms.
69 // AVG64 on accel: hardware averages 64 internal samples, yielding a
70 // noise-reduced output; effective anti-aliasing bandwidth = ODR/4 = 50 Hz,
71 // which keeps the 30 Hz digital cutoff well within the passband.
72 // Gyro AVG1: no averaging for lowest latency; hardware BW = ODR/2 = 100 Hz.
73 config[0].cfg.acc.odr = BMI3_ACC_ODR_200HZ;
74 config[0].cfg.acc.range = BMI3_ACC_RANGE_16G;
75 config[0].cfg.acc.bwp = BMI3_ACC_BW_ODR_QUARTER;
76 config[0].cfg.acc.avg_num = BMI3_ACC_AVG64;
77 config[0].cfg.acc.acc_mode = BMI3_ACC_MODE_NORMAL;
78
79 config[1].cfg.gyr.odr = BMI3_GYR_ODR_200HZ;
80 config[1].cfg.gyr.range = BMI3_GYR_RANGE_2000DPS;
81 config[1].cfg.gyr.bwp = BMI3_GYR_BW_ODR_HALF;
82 config[1].cfg.gyr.gyr_mode = BMI3_GYR_MODE_NORMAL;
83 config[1].cfg.gyr.avg_num = BMI3_GYR_AVG1;
84
85 if (bmi323_set_sensor_config(config, 2, &device) != BMI3_OK)
86 return 5;
87
88 _sensor_active = true;
89 return 0;
90}

92int SIMU_BMI323::Update()
93{
94 int rslt = BMI323_OK;
95
96 if(!_sensor_active)
97 {
98 return -999;
99 }
100
101 uint16_t int_status = 0;
102
103 rslt = bmi323_get_int1_status(&int_status, &device);
104 if (rslt != BMI3_OK)
105 return rslt;
106
107 // ODR is 200 Hz (new data every 5 ms); sensor task polls every 2 ms.
108 // Return early on calls where data isn't ready yet rather than
109 // spinning and blocking the FreeRTOS scheduler.
110 if (!((int_status & BMI3_INT_STATUS_ACC_DRDY) &&
111 (int_status & BMI3_INT_STATUS_GYR_DRDY) &&
112 (int_status & BMI3_INT_STATUS_TEMP_DRDY)))
113 {
114 return BMI3_OK;
115 }
116
117 bmi3_sensor_data sensor_data[3] = {};
118 sensor_data[0].type = BMI323_ACCEL;
119 sensor_data[1].type = BMI323_GYRO;
120 sensor_data[2].type = BMI323_TEMP;
121
122 rslt = bmi323_get_sensor_data(sensor_data, 3, &device);
123 if (rslt != BMI3_OK)
124 return rslt;
125
126 _raw.accel_linear.x = lsb_to_g(sensor_data[0].sens_data.acc.x, 16.0f, device.resolution);
127 _raw.accel_linear.y = lsb_to_g(sensor_data[0].sens_data.acc.y, 16.0f, device.resolution);
128 _raw.accel_linear.z = lsb_to_g(sensor_data[0].sens_data.acc.z, 16.0f, device.resolution);
129
130 _raw.ang_vel.x = lsb_to_dps(sensor_data[1].sens_data.gyr.x, 2000.0f, device.resolution);
131 _raw.ang_vel.y = lsb_to_dps(sensor_data[1].sens_data.gyr.y, 2000.0f, device.resolution);
132 _raw.ang_vel.z = lsb_to_dps(sensor_data[1].sens_data.gyr.z, 2000.0f, device.resolution);
133
134 float tempForDBG =
135 (float)(((float)((int16_t)sensor_data[2].sens_data.temp.temp_data) / 512.0f) + 23.0f);
136
137 return rslt;
138}
