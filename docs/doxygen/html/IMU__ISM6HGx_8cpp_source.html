26int8_t IMU_ISM6HGx::Init() {
27 /* Initialize mems driver interface */
28 dev_ctx.write_reg = ISM_SPI_Write;
29 dev_ctx.read_reg = ISM_SPI_Read;
30 dev_ctx.mdelay = ISM_platform_MSdelay;
31 dev_ctx.handle = &SENSOR_BUS;
32
33 /* Init test platform */
34 Platform_Init();
35
36 /* Perform device power-on-reset */
37 ism6hg256x_sw_por(&dev_ctx);
38
39 /* Wait sensor boot time */
40 ISM_platform_MSdelay(10);
41
42 /* Check device ID */
43 ism6hg256x_device_id_get(&dev_ctx, &whoamI);
44 if (whoamI != ISM6HG256X_ID)
45 return -1;
46
47 /* Enable Block Data Update */
48 ism6hg256x_block_data_update_set(&dev_ctx, PROPERTY_ENABLE);
49
50 /* Set Output Data Rate */
51 ism6hg256x_xl_setup(&dev_ctx, ISM6HG256X_ODR_AT_60Hz, ISM6HG256X_XL_HIGH_PERFORMANCE_MD);
52 ism6hg256x_hg_xl_data_rate_set(&dev_ctx, ISM6HG256X_HG_XL_ODR_AT_960Hz, 1);
53 ism6hg256x_gy_setup(&dev_ctx, ISM6HG256X_ODR_AT_120Hz, ISM6HG256X_GY_HIGH_PERFORMANCE_MD);
54
55 /* Set full scale */
56 ism6hg256x_xl_full_scale_set(&dev_ctx, ISM6HG256X_2g);
57 ism6hg256x_hg_xl_full_scale_set(&dev_ctx, ISM6HG256X_256g);
58 ism6hg256x_gy_full_scale_set(&dev_ctx, ISM6HG256X_2000dps);
59
60 /* Configure filtering chain */
61 filt_settling_mask.drdy = PROPERTY_ENABLE;
62 filt_settling_mask.irq_xl = PROPERTY_ENABLE;
63 filt_settling_mask.irq_g = PROPERTY_ENABLE;
64 ism6hg256x_filt_settling_mask_set(&dev_ctx, filt_settling_mask);
65 ism6hg256x_filt_gy_lp1_set(&dev_ctx, PROPERTY_ENABLE);
66 ism6hg256x_filt_gy_lp1_bandwidth_set(&dev_ctx, ISM6HG256X_GY_ULTRA_LIGHT);
67 ism6hg256x_filt_xl_lp2_set(&dev_ctx, PROPERTY_ENABLE);
68 ism6hg256x_filt_xl_lp2_bandwidth_set(&dev_ctx, ISM6HG256X_XL_STRONG);
69
70 /* Init accumulators */
71 lowg_xl_sum[0] = lowg_xl_sum[1] = lowg_xl_sum[2] = 0.0f;
72 hg_xl_sum[0] = hg_xl_sum[1] = hg_xl_sum[2] = 0.0f;
73 gyro_sum[0] = gyro_sum[1] = gyro_sum[2] = 0.0f;
74 temp_sum = 0.0f;
75
76 lowg_xl_cnt = 0;
77 hg_xl_cnt = 0;
78 gyro_cnt = 0;
79 temp_cnt = 0;
80
81 /* Enable interrupt on High-G XL DRDY -> INT1 */
82 pin_int = {0};
83 pin_int.drdy_hg_xl = PROPERTY_ENABLE;
84 ism6hg256x_pin_int1_route_hg_set(&dev_ctx, &pin_int);
85
86 return 0;
87}

89int8_t IMU_ISM6HGx::Update()
90{
91 // Consume pending interrupt flag
92 if (thread_wake) {
93 ism6hg256x_data_ready_t status;
94
95 thread_wake = 0;
96
97 ism6hg256x_flag_data_ready_get(&dev_ctx, &status);
98
99 /* Low-G accelerometer */
100 if (status.drdy_xl) {
101 ism6hg256x_acceleration_raw_get(&dev_ctx, data_raw_motion);
102
103 lowg_xl_sum[0] += ism6hg256x_from_fs2_to_mg(data_raw_motion[0]);
104 lowg_xl_sum[1] += ism6hg256x_from_fs2_to_mg(data_raw_motion[1]);
105 lowg_xl_sum[2] += ism6hg256x_from_fs2_to_mg(data_raw_motion[2]);
106 lowg_xl_cnt++;
107 }
108
109 /* High-G accelerometer */
110 if (status.drdy_hgxl) {
111 ism6hg256x_hg_acceleration_raw_get(&dev_ctx, data_raw_motion);
112
113 hg_xl_sum[0] += ism6hg256x_from_fs256_to_mg(data_raw_motion[0]);
114 hg_xl_sum[1] += ism6hg256x_from_fs256_to_mg(data_raw_motion[1]);
115 hg_xl_sum[2] += ism6hg256x_from_fs256_to_mg(data_raw_motion[2]);
116 hg_xl_cnt++;
117 }
118
119 /* Gyroscope */
120 if (status.drdy_gy) {
121 ism6hg256x_angular_rate_raw_get(&dev_ctx, data_raw_motion);
122
123 gyro_sum[0] += ism6hg256x_from_fs2000_to_mdps(data_raw_motion[0]);
124 gyro_sum[1] += ism6hg256x_from_fs2000_to_mdps(data_raw_motion[1]);
125 gyro_sum[2] += ism6hg256x_from_fs2000_to_mdps(data_raw_motion[2]);
126 gyro_cnt++;
127 }
128
129 /* Temperature */
130 if (status.drdy_temp) {
131 ism6hg256x_temperature_raw_get(&dev_ctx, &data_raw_temperature);
132 temp_sum += ism6hg256x_from_lsb_to_celsius(data_raw_temperature);
133 temp_cnt++;
134 }
135 }
136
137 // Output averaged data once enough low-G samples have accumulated
138 // (low-G is the lowest ODR at 60 Hz, so it is the pacing reference)
139 if (lowg_xl_cnt >= CNT_FOR_OUTPUT) {
140
141 // Low-G accelerometer average (mg)
142 data.acceleration.x = lowg_xl_sum[0] / lowg_xl_cnt;
143 data.acceleration.y = lowg_xl_sum[1] / lowg_xl_cnt;
144 data.acceleration.z = lowg_xl_sum[2] / lowg_xl_cnt;
145 lowg_xl_sum[0] = lowg_xl_sum[1] = lowg_xl_sum[2] = 0.0f;
146 lowg_xl_cnt = 0;
147
148 // High-G accelerometer average (mg)
149 if (hg_xl_cnt > 0) {
150 data.accelerationHighG.x = hg_xl_sum[0] / hg_xl_cnt;
151 data.accelerationHighG.y = hg_xl_sum[1] / hg_xl_cnt;
152 data.accelerationHighG.z = hg_xl_sum[2] / hg_xl_cnt;
153 hg_xl_sum[0] = hg_xl_sum[1] = hg_xl_sum[2] = 0.0f;
154 hg_xl_cnt = 0;
155 }
156
157 // Gyroscope average (mdps)
158 if (gyro_cnt > 0) {
159 data.angular_velocity.x = gyro_sum[0] / gyro_cnt;
160 data.angular_velocity.y = gyro_sum[1] / gyro_cnt;
161 data.angular_velocity.z = gyro_sum[2] / gyro_cnt;
162 gyro_sum[0] = gyro_sum[1] = gyro_sum[2] = 0.0f;
163 gyro_cnt = 0;
164 }
165
166 // Temperature average (°C)
167 if (temp_cnt > 0) {
168 data.temperature_degC = temp_sum / temp_cnt;
169 temp_sum = 0.0f;
170 temp_cnt = 0;
171 }
172
173 return 0; // new averaged data is ready
174 }
175
176 return 1; // still accumulating data
177}
