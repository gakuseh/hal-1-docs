17SAsym<float> Servo_Axon_Mini_MKII::readRawVoltage() {
18 // ADC3 is configured in scan mode (NbrOfConversion=2):
19 // Rank 1 = CH0 (PC2_C, S1 encoder)
20 // Rank 2 = CH1 (PC3_C, S2 encoder)
21 // Start once, poll twice — no channel reconfiguration needed.
22 HAL_ADC_Start(&hadc3);
23
24 uint32_t raw1 = 0, raw2 = 0;
25 if (HAL_ADC_PollForConversion(&hadc3, 5) == HAL_OK)
26 raw1 = HAL_ADC_GetValue(&hadc3);
27 if (HAL_ADC_PollForConversion(&hadc3, 5) == HAL_OK)
28 raw2 = HAL_ADC_GetValue(&hadc3);
29
30 HAL_ADC_Stop(&hadc3);
31
32 const float v1 = (3.3f * static_cast<float>(raw1)) / 4095.0f;
33 const float v2 = (3.3f * static_cast<float>(raw2)) / 4095.0f;
34
35 // Exponential moving average — α = 0.1 (10% new sample, 90% history).
36 // Initialised to mid-rail (1.65 V) so the first real sample blends in smoothly.
37 static float s1 = 1.65f, s2 = 1.65f;
38 constexpr float alpha = 0.5f;
39 s1 = alpha * v1 + (1.0f - alpha) * s1;
40 s2 = alpha * v2 + (1.0f - alpha) * s2;
41
42 return {s1, s2};
43}

49SAsym<float> Servo_Axon_Mini_MKII::readCurrentAngle() {
50 SAsym<float> v = readRawVoltage();
51 return {
52 (v.S1 - _zeroVoltage.S1) * (360.0f / 3.3f),
53 (v.S2 - _zeroVoltage.S2) * (360.0f / 3.3f)
54 };

64SAsym<float> Servo_Axon_Mini_MKII::calculateError() {
65 SAsym<float> value = readCurrentAngle();
66
67 // Guard against divide by zero when trackedAngle is near 0
68 float denom1 = fabsf(data.trackedAngle.S1) > 0.001f ? fabsf(data.trackedAngle.S1) : 1.0f;
69 float denom2 = fabsf(data.trackedAngle.S2) > 0.001f ? fabsf(data.trackedAngle.S2) : 1.0f;
70
71 SAsym<float> output = {
72 (value.S1 - data.trackedAngle.S1) / denom1,
73 (value.S2 - data.trackedAngle.S2) / denom2
74 };
75 return output;
76}

81void Servo_Axon_Mini_MKII::Actuate(SAsym<float> input) {
82 float sf = 5.55555555556f; // µs per degree
83
84 float us1 = 1500.0f + (input.S1 + config.AngleOffsetDEGREES.S1) * sf;
85 float us2 = 1500.0f + (input.S2 + config.AngleOffsetDEGREES.S2) * sf;
86
87 // 1µs = 2 counts @ 2MHz
88 uint32_t ccr1 = (uint32_t)(us1 * 2.0f);
89 uint32_t ccr2 = (uint32_t)(us2 * 2.0f);
90
91 TIM3->CCR1 = ccr1;
92 TIM3->CCR2 = ccr2;
93}

112bool Servo_Axon_Mini_MKII::Init(SAsym<float> AngOffset,
113 PRECISION precision,
114 bool debug) {
115 CFG_Axon_Mini_MKII configDesired;
116 configDesired.AngleOffsetDEGREES = AngOffset;
117 configDesired.Precision = precision;
118 configDesired.ENABLE_DEBUG = debug;
119 config = configDesired;
120
121 // Calibrate ADC before any reads
122 HAL_ADCEx_Calibration_Start(&hadc3, ADC_CALIB_OFFSET, ADC_SINGLE_ENDED);
123
124 // Start TIM3 PWM channels
125 HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_1);
126 HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_2);
127
128 // Drive to mechanical center, let servo settle, then capture zero reference.
129 // _zeroVoltage defaults to 1.65V so readCurrentAngle() is valid during the settle wait.
130 Actuate({0.0f, 0.0f});
131 vTaskDelay(pdMS_TO_TICKS(200));
132 SAsym<float> zv = readRawVoltage();
133 _zeroVoltage.S1 = zv.S1;
134 _zeroVoltage.S2 = zv.S2;
135
136 // Tolerance based on precision setting
137 float tolerance = 0.0f;
138 switch (config.Precision) {
139 case PRECISION::ONE_DEGREE: tolerance = 1.0f; break;
140 case PRECISION::TENTH_DEGREE: tolerance = 0.1f; break;
141 case PRECISION::HUNDREDTH_DEGREE: tolerance = 0.01f; break;
142 case PRECISION::THOUSANTHS_DEGREE: tolerance = 0.001f; break;
143 default: tolerance = 0.1f; break;
144 }
145
146 // Target is 0deg + offset for both motors
147 float target1 = config.AngleOffsetDEGREES.S1;
148 float target2 = config.AngleOffsetDEGREES.S2;
149
150 const TickType_t timeout = pdMS_TO_TICKS(3000); // 3s total
151 const TickType_t poll_rate = pdMS_TO_TICKS(5); // poll at 200Hz
152 const TickType_t start = xTaskGetTickCount();
153
154 TickType_t last_wake = xTaskGetTickCount();
155
156 while ((xTaskGetTickCount() - start) < timeout) {
157 data.currentAngle = readCurrentAngle();
158
159 float err1 = fabsf(data.currentAngle.S1 - target1);
160 float err2 = fabsf(data.currentAngle.S2 - target2);
161
162 data.Nominal.S1 = (err1 <= tolerance);
163 data.Nominal.S2 = (err2 <= tolerance);
164
165 if (data.Nominal.S1 && data.Nominal.S2) {
166 // Both motors at target — stop and return success
167 Actuate({target1, target2});
168 return true;
169 }
170
171 // Drive toward target
172 Actuate({target1, target2});
173
174 // Yield to RTOS scheduler, wake at precise interval
175 vTaskDelayUntil(&last_wake, poll_rate);
176 }
177
178 // Timeout — record final error state
179 data.currentAngle = readCurrentAngle();
180 data.trackedError.S1 = fabsf(data.currentAngle.S1 - target1);
181 data.trackedError.S2 = fabsf(data.currentAngle.S2 - target2);
182 data.Nominal.S1 = (data.trackedError.S1 <= tolerance);
183 data.Nominal.S2 = (data.trackedError.S2 <= tolerance);
184
185 return data.Nominal.S1 && data.Nominal.S2;
186}

198void Servo_Axon_Mini_MKII::Update(float S1_Target_Deg, float S2_Target_Deg) {
199 data.targetAngle = {S1_Target_Deg, S2_Target_Deg};
200 data.trackedAngle = {S1_Target_Deg, S2_Target_Deg};
201
202 // Open-loop: Actuate() maps angle→PWM directly via the 1500µs center formula.
203 // Closed-loop correction is disabled because readCurrentAngle() returns
204 // absolute encoder position in 0–360° while targets are deflection angles
205 // in a different coordinate frame. Without encoder calibration the error
206 // term is meaningless and drives large spurious corrections causing jitter.
207 Actuate({S1_Target_Deg, S2_Target_Deg});
208
209 data.currentAngle = readCurrentAngle();
210 data.smoothedCorrection = {0.0f, 0.0f};
211
212 // trackedError: raw encoder deviation from target (informational only)
213 data.trackedError.S1 = fabsf(data.currentAngle.S1 - S1_Target_Deg);
214 data.trackedError.S2 = fabsf(data.currentAngle.S2 - S2_Target_Deg);
215}

222void Servo_Axon_Mini_MKII::SetOffset(SAsym<float> offset) {
223 config.AngleOffsetDEGREES = offset;
224 // Re-actuate at the last commanded target so the servo immediately moves
225 // to reflect the new trim even when the CTRLs task is not running.
226 Actuate(data.targetAngle);
