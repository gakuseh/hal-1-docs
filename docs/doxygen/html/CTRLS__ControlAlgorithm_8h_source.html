30template<int R, int C>
31struct Mat {
32 float d[R][C];
33
35 Mat() { memset(d, 0, sizeof(d)); }
37 float& operator()(int r, int c) { return d[r][c]; }
39 const float& operator()(int r, int c) const { return d[r][c]; }
40};

43template<int R, int K, int C>
44inline Mat<R,C> operator*(const Mat<R,K>& A, const Mat<K,C>& B) {
45 Mat<R,C> out;
46 for(int r=0;r<R;r++) for(int c=0;c<C;c++) for(int k=0;k<K;k++)
47 out(r,c) += A(r,k)*B(k,c);
48 return out;
49}
50
52template<int R, int C>
53inline Mat<R,C> operator+(const Mat<R,C>& A, const Mat<R,C>& B) {
54 Mat<R,C> out;
55 for(int r=0;r<R;r++) for(int c=0;c<C;c++) out(r,c) = A(r,c)+B(r,c);
56 return out;
57}
58
60template<int R, int C>
61inline Mat<R,C> operator-(const Mat<R,C>& A, const Mat<R,C>& B) {
62 Mat<R,C> out;
63 for(int r=0;r<R;r++) for(int c=0;c<C;c++) out(r,c) = A(r,c)-B(r,c);
64 return out;
65}
66
68template<int R, int C>
69inline Mat<C,R> Tr(const Mat<R,C>& A) {
70 Mat<C,R> out;
71 for(int r=0;r<R;r++) for(int c=0;c<C;c++) out(c,r) = A(r,c);
72 return out;
73}
