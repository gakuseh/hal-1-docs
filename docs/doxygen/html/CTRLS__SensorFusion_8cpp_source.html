11void EKF::reset(const RocketConfig& cfg) {
12 // Initial state from config
13 for (int i = 0; i < 10; i++) xhat(i,0) = cfg.x0[i];
14
15 // Initial covariance: scaled identity
16 P_ = cfg.P0_scale * eye<10>();
17
18 // Process noise Q (diagonal)
19 Q_ = Mat<10,10>();
20 for (int i = 0; i < 3; i++) Q_(i, i) = cfg.Q_angular[i];
21 for (int i = 0; i < 3; i++) Q_(3+i, 3+i) = cfg.Q_velocity[i];
22 for (int i = 0; i < 4; i++) Q_(6+i, 6+i) = cfg.Q_quat[i];
23
24 // Measurement noise R (diagonal 6×6)
25 R_ = Mat<6,6>();
26 const float accel_var = cfg.accel_noise_std_g * cfg.accel_noise_std_g;
27 const float gyro_var = cfg.gyro_noise_std * cfg.gyro_noise_std;
28 for (int i = 0; i < 3; i++) R_(i, i) = accel_var;
29 for (int i = 0; i < 3; i++) R_(3+i, 3+i) = gyro_var;
30}

33void EKF::predict_(float t, float dt, float u, float alt, StateMat& A_out) {
34 if (dt <= 0.0f || dt > 0.5f) return;
35
36 // Euler integration
37 StateVec xdot = phys_.eom(t, xhat, u, alt);
38 for (int i = 0; i < 10; i++) xhat(i,0) += xdot(i,0) * dt;
39
40 normalizeQuaternion_();
41
42 // Linearise at updated state
43 A_out = phys_.jacobianA(t, xhat, u, alt);
44
45 // Discrete F = I + A*dt
46 StateMat F = eye<10>() + dt * A_out;
47
48 // P = F*P*F^T + Q
49 P_ = F * P_ * Tr(F) + Q_;
50}

52// ─── correct_ ─────────────────────────────────────────────────────────────────
53void EKF::correct_(float t, float u, const MeasVec& y_meas,
54 float alt, const StateMat& A,
55 bool burning) {
56 // Build effective R (inflate accel during burn)
57 Mat<6,6> R_eff = R_;
58 if (burning) {
59 const float bv = cfg_.accel_burn_std_g * cfg_.accel_burn_std_g;
60 for (int i = 0; i < 3; i++) R_eff(i,i) = bv;
61 }
62
63 // Measurement Jacobian
64 MeasMat C = phys_.jacobianC(t, xhat, u, A, alt);
65
66 // Innovation: y_pred from noiseless sensor model
67 MeasVec y_pred = phys_.predictMeasurement(t, xhat, u, alt);
68 MeasVec innov;
69 for (int i = 0; i < 6; i++) innov(i,0) = y_meas(i,0) - y_pred(i,0);
70
71 // Innovation covariance S = C*P*C^T + R_eff
72 Mat<6,6> S = C * P_ * Tr(C) + R_eff;
73
74 // Kalman gain K = P*C^T*S^-1 (10×6)
75 Mat<10,6> K = P_ * Tr(C) * inv6(S);
76
77 // Gain limiting per spec:
78 // - Zero accel columns (0-2) for rate rows (0-2) and quat rows (6-9)
79 // - Zero gyro columns (3-5) for velocity rows (3-5) and quat rows (6-9)
80 // - Zero all accel columns during burn
81 for (int j = 0; j < 3; j++) {
82 for (int i = 0; i < 3; i++) K(i, j) = 0.0f; // rate rows, accel cols
83 for (int i = 6; i < 10; i++) K(i, j) = 0.0f; // quat rows, accel cols
84 if (burning)
85 for (int i = 0; i < 10; i++) K(i, j) = 0.0f; // all rows, accel cols
86 }
87 for (int j = 3; j < 6; j++) {
88 for (int i = 3; i < 6; i++) K(i, j) = 0.0f; // vel rows, gyro cols
89 for (int i = 6; i < 10; i++) K(i, j) = 0.0f; // quat rows, gyro cols
90 }
91
92 // State update
93 StateVec dx = K * innov;
94 for (int i = 0; i < 10; i++) xhat(i,0) += dx(i,0);
95
96 // Covariance update (Joseph form: guarantees PSD)
97 StateMat IKC = eye<10>() - K * C;
98 P_ = IKC * P_ * Tr(IKC) + K * R_eff * Tr(K);
99}

102StateVec EKF::update(float t, float dt, const MeasVec& y_meas, float u,
103 float alt) {
104 const bool on_rail = (t < cfg_.t_rail);
105 const bool burning = (t >= 0.0f && t < cfg_.t_burnout);
106
107 if (on_rail) {
108 applyRailConstraint_();
109 StateMat A;
110 predict_(t, dt, u, alt, A);
111 applyRailConstraint_();
112 correct_(t, u, y_meas, alt, A, burning);
113 applyRailConstraint_();
114 } else {
115 StateMat A;
116 predict_(t, dt, u, alt, A);
117 correct_(t, u, y_meas, alt, A, burning);
118 }
119
120 normalizeQuaternion_();
121 return xhat;
122}

131void EKF::applyRailConstraint_() {
132 // Zero pitch/yaw rates and lateral velocities; lock quaternion to initial
133 xhat(0,0) = 0.0f; // w1 (pitch rate)
134 xhat(1,0) = 0.0f; // w2 (yaw rate)
135 xhat(3,0) = 0.0f; // v1 (lateral)
136 xhat(4,0) = 0.0f; // v2 (lateral)
137 if (xhat(5,0) < 0.0f) xhat(5,0) = 0.0f; // v3 >= 0 on rail
138 // Lock quaternion to initial orientation
139 xhat(6,0) = cfg_.x0[6];
140 xhat(7,0) = cfg_.x0[7];
141 xhat(8,0) = cfg_.x0[8];
142 xhat(9,0) = cfg_.x0[9];
143}

145void EKF::normalizeQuaternion_() {
146 const float n = sqrtf(xhat(6,0)*xhat(6,0) + xhat(7,0)*xhat(7,0)
147 + xhat(8,0)*xhat(8,0) + xhat(9,0)*xhat(9,0));
148 if (n > 1e-8f) {
149 xhat(6,0) /= n; xhat(7,0) /= n; xhat(8,0) /= n; xhat(9,0) /= n;
150 } else {
151 xhat(6,0) = 1.0f; xhat(7,0) = 0.0f; xhat(8,0) = 0.0f; xhat(9,0) = 0.0f;
152 }
