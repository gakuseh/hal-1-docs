35uint8_t Barometer::Init()
36{
37 spi_intf.spi_handle = _spi;
38
39 /* Map sensor index to chip select pins */
40 switch(InitDev)
41 {
42 case SENSOR1_I:
43 spi_intf.cs_port = BMP390_CS1_GPIO_Port;
44 spi_intf.cs_pin = BMP390_CS1_Pin;
45 break;
46
47 case SENSOR2_I:
48 spi_intf.cs_port = BMP390_CS2_GPIO_Port;
49 spi_intf.cs_pin = BMP390_CS2_Pin;
50 break;
51
52 case SENSOR3_I:
53 spi_intf.cs_port = BMP390_CS3_GPIO_Port;
54 spi_intf.cs_pin = BMP390_CS3_Pin;
55 break;
56
57 default:
58 return 1;
59 }
60
61 device.intf = BMP3_SPI_INTF;
62 device.intf_ptr = &spi_intf;
63
64 device.read = BMP_SPI_Read;
65 device.write = BMP_SPI_Write;
66 device.delay_us = BMP_delay_us;
67
68
69 if(bmp3_init(&device) != BMP3_OK)
70 return 2;
71
72 /* Sensor configuration */
73 settings.press_en = BMP3_ENABLE;
74 settings.temp_en = BMP3_ENABLE;
75 settings.odr_filter.press_os = BMP3_OVERSAMPLING_2X;
76 settings.odr_filter.temp_os = BMP3_OVERSAMPLING_2X;
77 settings.odr_filter.odr = BMP3_ODR_100_HZ;
78
79 settings_sel = BMP3_SEL_PRESS_EN | BMP3_SEL_TEMP_EN | BMP3_SEL_PRESS_OS | BMP3_SEL_TEMP_OS | BMP3_SEL_ODR | BMP3_SEL_DRDY_EN;
80
81 if(bmp3_set_sensor_settings(
82 settings_sel,
83 &settings,
84 &device) != BMP3_OK)
85 {
86 return 3;
87 }
88
89 // now set operation mode
90 settings.op_mode = BMP3_MODE_NORMAL;
91 if(bmp3_set_op_mode(&settings, &device) != BMP3_OK)
92 return 4;
93
94 _sensor_active = true;
95
96 return 0;
97}

103uint8_t Barometer::Update()
104{
105 if(!_sensor_active || (bmp3_get_status(&status, &device) != BMP3_OK))
106 return 1;
107
108 if (status.intr.drdy != BMP3_ENABLE)
109 return 0; // no new sample yet — retain last valid _raw
110
111 bmp3_data data = { 0 };
112
113 if (bmp3_get_sensor_data(BMP3_PRESS_TEMP, &data, &device) != BMP3_OK)
114 return 2;
115 if (bmp3_get_status(&status, &device) != BMP3_OK)
116 return 3;
117
118 if (data.pressure < BARO_PRESS_MIN_PA || data.pressure > BARO_PRESS_MAX_PA)
119 return 0; // implausible reading — retain last valid _raw
120
121 _raw.Pressure = static_cast<float>(data.pressure);
122 _raw.Temperature = static_cast<float>(data.temperature);
123 _raw.heightMeters =
124 44330.0F *
125 (1.0f - powf(static_cast<float>(data.pressure) / SEA_LEVEL_PRESSURE, 0.1903f));
126
127 return 0;
128}
