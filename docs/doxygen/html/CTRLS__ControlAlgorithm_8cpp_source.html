251// ─── inv6: LU decomposition with partial pivoting ────────────────────────────
252Mat<6,6> inv6(const Mat<6,6>& A) {
253 Mat<6,6> LU = A;
254 int piv[6] = {0,1,2,3,4,5};
255
256 for (int k = 0; k < 6; k++) {
257 // partial pivot
258 int mx = k;
259 float mxv = fabsf(LU(k,k));
260 for (int i = k+1; i < 6; i++) {
261 if (fabsf(LU(i,k)) > mxv) { mxv = fabsf(LU(i,k)); mx = i; }
262 }
263 if (mx != k) {
264 for (int j = 0; j < 6; j++) { float t = LU(k,j); LU(k,j)=LU(mx,j); LU(mx,j)=t; }
265 int tp = piv[k]; piv[k]=piv[mx]; piv[mx]=tp;
266 }
267 if (fabsf(LU(k,k)) < 1e-12f) return eye<6>(); // singular fallback
268 for (int i = k+1; i < 6; i++) {
269 LU(i,k) /= LU(k,k);
270 for (int j = k+1; j < 6; j++) LU(i,j) -= LU(i,k)*LU(k,j);
271 }
272 }
273
274 Mat<6,6> inv;
275 for (int col = 0; col < 6; col++) {
276 float b[6] = {};
277 b[col] = 1.0f;
278 // apply row permutation
279 for (int k = 0; k < 6; k++) {
280 if (piv[k] != k) { float t = b[piv[k]]; b[piv[k]]=b[k]; b[k]=t; }
281 }
282 // forward substitution (unit lower triangular L)
283 for (int i = 1; i < 6; i++)
284 for (int j = 0; j < i; j++) b[i] -= LU(i,j)*b[j];
285 // back substitution (upper triangular U)
286 for (int i = 5; i >= 0; i--) {
287 for (int j = i+1; j < 6; j++) b[i] -= LU(i,j)*b[j];
288 b[i] /= LU(i,i);
289 }
290 for (int i = 0; i < 6; i++) inv(i,col) = b[i];
291 }
292 return inv;
293}

295// ─── lerp ─────────────────────────────────────────────────────────────────────
296float lerp(const float* xs, const float* ys, int n, float xi) {
297 if (n <= 0) return 0.0f;
298 if (xi <= xs[0]) return ys[0];
299 if (xi >= xs[n-1]) return ys[n-1];
300 // binary search for bracket
301 int lo = 0, hi = n-1;
302 while (hi - lo > 1) {
303 int mid = (lo+hi) >> 1;
304 if (xs[mid] <= xi) lo = mid; else hi = mid;
305 }
306 float t = (xi - xs[lo]) / (xs[hi] - xs[lo]);
307 return ys[lo] + t * (ys[hi] - ys[lo]);
308}

313 const float gamma = (cfg.Cr > 0.0f) ? (cfg.Ct / cfg.Cr) : 0.0f;
314 r_t = cfg.d * 0.5f;
315 tau = (r_t > 0.0f) ? ((cfg.s + r_t) / r_t) : 1.0f;
316
317 y_ma = (cfg.s / 3.0f) * (1.0f + 2.0f*gamma) / (1.0f + gamma);
318
319 const float tau2 = tau * tau;
320 const float tau_minus_1 = tau - 1.0f;
321 const float asin_arg = (tau2 - 1.0f) / (tau2 + 1.0f);
322 const float asin_term = asinf(fmaxf(-1.0f, fminf(1.0f, asin_arg)));
323 const float shape_f =
324 (1.0f / (M_PI * M_PI)) *
325 ((M_PI*M_PI / 4.0f) * ((tau + 1.0f)*(tau + 1.0f) / tau2)
326 + (M_PI * (tau2 + 1.0f)*(tau2 + 1.0f) / (tau2 * tau_minus_1*tau_minus_1)) * asin_term
327 - (2.0f * M_PI * (tau + 1.0f)) / (tau * tau_minus_1)
328 + ((tau2 + 1.0f)*(tau2 + 1.0f) / (tau2 * tau_minus_1*tau_minus_1)) * asin_term*asin_term
329 - (4.0f * (tau + 1.0f) / (tau * tau_minus_1)) * asin_term
330 + (8.0f / (tau_minus_1*tau_minus_1)) * logf((tau2 + 1.0f) / (2.0f * tau)));
331
332 // Runtime uses M_f_roll = K_f * q_dyn.
333 K_f = shape_f * (float)cfg.N_fins * (y_ma + r_t) * cfg.Cnalpha_fin
334 * cfg.delta_rad * cfg.A_ref;
335
336 const float trap_integral = cfg.s / 12.0f *
337 ((cfg.Cr + 3.0f*cfg.Ct) * cfg.s*cfg.s
338 + 4.0f * (cfg.Cr + 2.0f*cfg.Ct) * cfg.s * r_t
339 + 6.0f * (cfg.Cr + cfg.Ct) * r_t*r_t);
340 C_ldw = 2.0f * (float)cfg.N_fins * cfg.Cnalpha_fin
341 / (cfg.A_ref * cfg.d*cfg.d) * cosf(cfg.delta_rad) * trap_integral;
342 const float shape_d = 1.0f +
343 (((tau - gamma) / tau) - ((1.0f - gamma) / tau_minus_1) * logf(tau)) /
344 (((tau + 1.0f) * (tau - gamma) / 2.0f)
345 - ((1.0f - gamma) * (tau*tau*tau - 1.0f) / (3.0f * tau_minus_1)));
346
347 // Runtime uses M_d_roll = 0.5*rho*v_mag*K_d*w3.
348 K_d = shape_d * cfg.A_ref * cfg.d * C_ldw * (cfg.d * 0.5f);
349}

400WeatherSample Physics::weatherAtAltitude(float alt_m) const {
401 static constexpr int n = static_cast<int>(sizeof(kWeatherProfile) / sizeof(kWeatherProfile[0]));
402 if (n <= 0) {
403 WeatherSample fallback;
404 fallback.rho = cfg.rho_0;
405 fallback.temperature_K = 288.15f;
406 fallback.g = cfg.g_0;
407 fallback.wind_x = cfg.v_wind_x;
408 fallback.wind_y = cfg.v_wind_y;
409 return fallback;
410 }
411
412 auto sampleFrom = [](const WeatherPoint& p) {
413 WeatherSample s;
414 s.rho = p.rho;
415 s.temperature_K = p.temperature_K;
416 s.g = p.g;
417 s.wind_x = p.wind_x;
418 s.wind_y = p.wind_y;
419 return s;
420 };
421
422 if (alt_m <= kWeatherProfile[0].altitude_agl_m) return sampleFrom(kWeatherProfile[0]);
423 if (alt_m >= kWeatherProfile[n - 1].altitude_agl_m) return sampleFrom(kWeatherProfile[n - 1]);
424
425 int lo = 0;
426 int hi = n - 1;
427 while (hi - lo > 1) {
428 const int mid = (lo + hi) >> 1;
429 if (kWeatherProfile[mid].altitude_agl_m <= alt_m) lo = mid;
430 else hi = mid;
431 }
432
433 const WeatherPoint& a = kWeatherProfile[lo];
434 const WeatherPoint& b = kWeatherProfile[hi];
435 const float u = (alt_m - a.altitude_agl_m) / (b.altitude_agl_m - a.altitude_agl_m);
436
437 WeatherSample s;
438 s.rho = a.rho + u * (b.rho - a.rho);
439 s.temperature_K = a.temperature_K + u * (b.temperature_K - a.temperature_K);
440 s.g = a.g + u * (b.g - a.g);
441 s.wind_x = a.wind_x + u * (b.wind_x - a.wind_x);
442 s.wind_y = a.wind_y + u * (b.wind_y - a.wind_y);
443 return s;
444}

447void Physics::R_BW(float qw, float qx, float qy, float qz, float R[3][3]) {

449 R[0][0] = 1.0f - 2.0f*(qy2+qz2); R[0][1] = 2.0f*(qx*qy+qw*qz); R[0][2] = 2.0f*(qx*qz-qw*qy);
450 R[1][0] = 2.0f*(qx*qy-qw*qz); R[1][1] = 1.0f - 2.0f*(qx2+qz2); R[1][2] = 2.0f*(qy*qz+qw*qx);
451 R[2][0] = 2.0f*(qx*qz+qw*qy); R[2][1] = 2.0f*(qy*qz-qw*qx); R[2][2] = 1.0f - 2.0f*(qx2+qy2);
452}

485// ─── Full nonlinear EOM ───────────────────────────────────────────────────────
486StateVec Physics::eom(float t, const StateVec& x, float u_zeta, float alt) const {
487 // Extract state
488 float w1=x(0,0), w2=x(1,0), w3=x(2,0);
489 float v1=x(3,0), v2=x(4,0), v3=x(5,0);
490 float qw=x(6,0), qx=x(7,0), qy=x(8,0), qz=x(9,0);
491
492 // Normalize quaternion
493 float qn = sqrtf(qw*qw+qx*qx+qy*qy+qz*qz);
494 if (qn > 1e-8f) { qw/=qn; qx/=qn; qy/=qn; qz/=qn; }
495
496 // Time-varying params
497 const float m = getMass(t);
498 const float I1 = getInertiaPitch(t);
499 const float I3 = getInertiaRoll(t);
500 const float I1d = getInertiaDotPitch(t);
501 const float I3d = getInertiaDotRoll(t);
502 const float xCG = getCG(t);
503 const float Thr = getThrust(t);
504 const WeatherSample wx = weatherAtAltitude(alt);
505
506 // Rotation matrix
507 float R[3][3];
508 R_BW(qw, qx, qy, qz, R);
509
510 // Air-relative velocity
511 float va[3];
512 airRelativeVelocity(x, wx.wind_x, wx.wind_y, va);
513
514 // Airspeed
515 const float vmag2 = va[0]*va[0]+va[1]*va[1]+va[2]*va[2];
516 const float vmag = sqrtf(vmag2);
517 static constexpr float EPS_AOA = 1.0e-9f;
518 const float vmag_reg = sqrtf(vmag2 + EPS_AOA*EPS_AOA);
519 const float v_xy_mag = sqrtf(va[0]*va[0] + va[1]*va[1] + EPS_AOA*EPS_AOA);
520
521 // Angle of attack (from axial z-body), clamped to ±15 deg
522 static constexpr float AoA_MAX = 15.0f * (M_PI / 180.0f);
523 float alpha = 0.0f;
524 if (vmag > 0.1f)
525 alpha = atan2f(v_xy_mag, va[2] + EPS_AOA);
526 if (alpha > AoA_MAX) alpha = AoA_MAX;
527 if (alpha < -AoA_MAX) alpha = -AoA_MAX;
528
529 // Directional canard-plane AoA, matching FV-Controls momentsforces.py.
530 const float canard_lat = va[0]*cfg.canard_dir_1 + va[1]*cfg.canard_dir_2;
531 const float canard_lat_mag = sqrtf(canard_lat*canard_lat + EPS_AOA*EPS_AOA);
532 const float vcan2 = canard_lat*canard_lat + va[2]*va[2];
533 const float vcan_mag = sqrtf(vcan2 + EPS_AOA*EPS_AOA);
534 float alpha_can = 0.0f;
535 if (vcan_mag > 0.1f)
536 alpha_can = atan2f(canard_lat_mag, va[2] + EPS_AOA);
537 if (alpha_can > AoA_MAX) alpha_can = AoA_MAX;
538 if (alpha_can < -AoA_MAX) alpha_can = -AoA_MAX;
539
540 // Mach, Cd, dynamic pressure
541 const float c_snd = speedOfSound(wx.temperature_K);
542 const float mach = (c_snd > 0.0f) ? vmag / c_snd : 0.0f;
543 const float Cd = getDragCoeff(mach, t);
544 const float rho = wx.rho;
545 const float q_dyn = 0.5f * rho * vmag2;
546
547 // Rail: no lift/moments before release. Thrust, gravity, and drag still act,
548 // matching FV-Controls' force model.
549 const bool on_rail = (t < cfg.t_rail);
550
551 // ── Forces (body frame) ──────────────────────────────────────────────────
552 float F[3] = {0.0f, 0.0f, 0.0f};
553
554 // Gravity: g_world = [0,0,-g]; g_body = R_BW * g_world = -g * R[:,2]
555 F[0] += m * (-wx.g * R[0][2]);
556 F[1] += m * (-wx.g * R[1][2]);
557 F[2] += m * (-wx.g * R[2][2]);
558
559 // Thrust along +z body axis
560 F[2] += Thr;
561
562 // Aerodynamic drag (along -v_air)
563 if (vmag > 0.1f) {
564 const float D = Cd * cfg.A_ref * q_dyn / vmag_reg;
565 F[0] -= D * va[0];
566 F[1] -= D * va[1];
567 F[2] -= D * va[2];
568 }
569
570 if (!on_rail && vmag > 0.1f) {
571 // Symmetric base-rocket normal force.
572 const float L_base = q_dyn * cfg.base_cn * alpha * cfg.A_ref;
573 const float cos_a = cosf(alpha);
574 F[0] += L_base * (-cos_a * va[0] / v_xy_mag);
575 F[1] += L_base * (-cos_a * va[1] / v_xy_mag);
576 F[2] += L_base * sinf(alpha);
577
578 // Two-canard normal force acts only in the configured canard plane.
579 const float q_can = 0.5f * rho * vcan2;
580 const float L_can = q_can * cfg.canard_cn * alpha_can * cfg.A_ref;
581 const float cos_a_can = cosf(alpha_can);
582 const float canard_lat_sign = canard_lat / canard_lat_mag;
583 F[0] += L_can * (-cos_a_can * canard_lat_sign * cfg.canard_dir_1);
584 F[1] += L_can * (-cos_a_can * canard_lat_sign * cfg.canard_dir_2);
585 F[2] += L_can * sinf(alpha_can);
586 }
587
588 // ── Moments (body frame) ─────────────────────────────────────────────────
589 float M[3] = {0.0f, 0.0f, 0.0f};
590
591 if (!on_rail) {
592 // Corrective moments: symmetric base plus directional canard plane.
593 if (vmag > 0.1f) {
594 const float cos_a = va[2] / vmag_reg;
595 const float C_raw_base = vmag2 * cfg.A_ref * cfg.base_cn * alpha
596 * (cfg.base_cp - xCG) * rho * 0.5f;
597 M[0] += -C_raw_base * cos_a * va[1] / v_xy_mag;
598 M[1] += C_raw_base * cos_a * va[0] / v_xy_mag;
599
600 const float cos_a_can = va[2] / vcan_mag;
601 const float C_raw_can = vcan2 * cfg.A_ref * cfg.canard_cn * alpha_can
602 * (cfg.canard_cp - xCG) * rho * 0.5f;
603 const float canard_lat_sign = canard_lat / canard_lat_mag;
604 M[0] += -C_raw_can * cos_a_can * canard_lat_sign * cfg.canard_dir_2;
605 M[1] += C_raw_can * cos_a_can * canard_lat_sign * cfg.canard_dir_1;
606 }
607
608 // Propulsive and aerodynamic pitch/yaw damping from FV-Controls.
609 const float mdot = (cfg.t_burnout > 0.0f) ? (cfg.m_prop / cfg.t_burnout) : 0.0f;
610 const float Cdp = (t < cfg.t_burnout)
611 ? mdot * (cfg.L_ne - xCG) * (cfg.L_ne - xCG)
612 : 0.0f;
613 const float Cda_base = 0.5f * rho * vmag_reg * cfg.A_ref
614 * cfg.base_cn * (cfg.base_cp - xCG) * (cfg.base_cp - xCG);
615 const float Cda_can = 0.5f * rho * vcan_mag * cfg.A_ref
616 * cfg.canard_cn * (cfg.canard_cp - xCG) * (cfg.canard_cp - xCG);
617 M[0] -= (Cdp + Cda_base + Cda_can * cfg.canard_dir_2 * cfg.canard_dir_2) * w1;
618 M[1] -= (Cdp + Cda_base + Cda_can * cfg.canard_dir_1 * cfg.canard_dir_1) * w2;
619
620 // Fin cant roll forcing (proportional to dynamic pressure)
621 M[2] += K_f * q_dyn;
622
623 // Roll aerodynamic damping: M = -0.5 * rho * v * K_d * w3
624 M[2] -= 0.5f * rho * vmag * K_d * w3;
625
626 // Canard roll moment (CFD numeric)
627 M[2] += canardMoment(vmag, u_zeta);
628 }
629
630 // ── Euler rigid-body (I1=I2 for symmetric rocket) ────────────────────────
631 const float I2 = I1;
632 StateVec xdot;
633 xdot(0,0) = ((I2-I3)*w2*w3 + M[0] - I1d*w1) / I1;
634 xdot(1,0) = ((I3-I1)*w3*w1 + M[1] - I1d*w2) / I1;
635 xdot(2,0) = ((I1-I2)*w1*w2 + M[2] - I3d*w3) / I3;
636
637 // ── Velocity: vdot = F/m - omega×v ───────────────────────────────────────
638 xdot(3,0) = F[0]/m - (w2*v3 - w3*v2);
639 xdot(4,0) = F[1]/m - (w3*v1 - w1*v3);
640 xdot(5,0) = F[2]/m - (w1*v2 - w2*v1);
641
642 // ── Quaternion kinematics: qdot = 0.5 * Omega(w) * q ─────────────────────
643 xdot(6,0) = 0.5f*(-w1*qx - w2*qy - w3*qz);
644 xdot(7,0) = 0.5f*( w1*qw + w3*qy - w2*qz);
645 xdot(8,0) = 0.5f*( w2*qw - w3*qx + w1*qz);
646 xdot(9,0) = 0.5f*( w3*qw + w2*qx - w1*qy);
647
648 return xdot;
649}

651// ─── Numerical Jacobian A = df/dx (central finite differences) ───────────────
652StateMat Physics::jacobianA(float t, const StateVec& x, float u, float alt) const {
653 static constexpr float EPS = 1e-5f;
654 StateMat A;
655 for (int j = 0; j < 10; j++) {
656 StateVec xp = x, xm = x;
657 xp(j,0) += EPS;
658 xm(j,0) -= EPS;
659 StateVec fp = eom(t, xp, u, alt);
660 StateVec fm = eom(t, xm, u, alt);
661 for (int i = 0; i < 10; i++)
662 A(i,j) = (fp(i,0) - fm(i,0)) / (2.0f * EPS);
663 }
664 return A;
665}

679// ─── Measurement Jacobian C = dy/dx (6×10) ───────────────────────────────────
680MeasMat Physics::jacobianC(float t, const StateVec& x, float u,
681 const StateMat& A, float alt) const {
682 const float w1=x(0,0), w2=x(1,0), w3=x(2,0);
683 const float v1=x(3,0), v2=x(4,0), v3=x(5,0);
684 const float qw=x(6,0), qx=x(7,0), qy=x(8,0), qz=x(9,0);
685 const WeatherSample wx = weatherAtAltitude(alt);
686 const float g_body_mag = wx.g;
687 const float accel_scale = cfg.g_0;
688
689 // J_cross = ∂(ω×v)/∂x [3×10]
690 // ω×v = [w2*v3-w3*v2, w3*v1-w1*v3, w1*v2-w2*v1]
691 Mat<3,10> Jcross;
692 // ∂/∂w1: [0, -v3, v2]
693 Jcross(0,0)=0.0f; Jcross(1,0)=-v3; Jcross(2,0)= v2;
694 // ∂/∂w2: [v3, 0, -v1]
695 Jcross(0,1)= v3; Jcross(1,1)=0.0f; Jcross(2,1)=-v1;
696 // ∂/∂w3: [-v2, v1, 0]
697 Jcross(0,2)=-v2; Jcross(1,2)= v1; Jcross(2,2)=0.0f;
698 // ∂/∂v1: [0, w3, -w2]
699 Jcross(0,3)=0.0f; Jcross(1,3)= w3; Jcross(2,3)=-w2;
700 // ∂/∂v2: [-w3, 0, w1]
701 Jcross(0,4)=-w3; Jcross(1,4)=0.0f; Jcross(2,4)= w1;
702 // ∂/∂v3: [w2, -w1, 0]
703 Jcross(0,5)= w2; Jcross(1,5)=-w1; Jcross(2,5)=0.0f;
704 // ∂/∂q = 0 (columns 6-9 already zero)
705
706 // J_g = ∂(g_body)/∂x [3×10]
707 // g_body = R_BW * [0,0,-g]; only q-columns nonzero
708 // R_BW col-2: [2(qx*qz-qw*qy), 2(qy*qz+qw*qx), 1-2(qx²+qy²)]
709 Mat<3,10> Jg;
710 // ∂g_body/∂qw
711 // g_body[0] = -g * 2*(qx*qz - qw*qy)
712 // ∂g_body[0]/∂qw = -g * 2*(-qy) = 2*g*qy
713 Jg(0,6) = 2.0f*g_body_mag*qy;
714 Jg(0,7) = -2.0f*g_body_mag*qz; // ∂/∂qx = -g*2*qz
715 Jg(0,8) = 2.0f*g_body_mag*qw; // ∂/∂qy = -g*2*(-qw) = 2*g*qw
716 Jg(0,9) = -2.0f*g_body_mag*qx; // ∂/∂qz = -g*2*qx
717
718 // g_body[1] = -g * 2*(qy*qz + qw*qx)
719 Jg(1,6) = -2.0f*g_body_mag*qx; // ∂/∂qw = -g*2*qx
720 Jg(1,7) = -2.0f*g_body_mag*qw; // ∂/∂qx = -g*2*qw
721 Jg(1,8) = -2.0f*g_body_mag*qz; // ∂/∂qy = -g*2*qz
722 Jg(1,9) = -2.0f*g_body_mag*qy; // ∂/∂qz = -g*2*qy
723
724 // g_body[2] = -g * (1 - 2*(qx²+qy²))
725 // ∂/∂qw = 0
726 Jg(2,7) = 4.0f*g_body_mag*qx; // ∂/∂qx = -g*(-4*qx) = 4*g*qx
727 Jg(2,8) = 4.0f*g_body_mag*qy; // ∂/∂qy = 4*g*qy
728
729 // Build C matrix
730 MeasMat C;
731
732 // Accel rows 0-2: C[0:3,:] = (A[3:6,:] + J_cross - J_g) / accel_scale
733 for (int i = 0; i < 3; i++) {
734 for (int j = 0; j < 10; j++) {
735 C(i, j) = (A(3+i, j) + Jcross(i, j) - Jg(i, j)) / accel_scale;
736 }
737 }
738 // Zero angular-rate columns on accel rows (accel doesn't directly observe w)
739 for (int i = 0; i < 3; i++)
740 for (int j = 0; j < 3; j++)
741 C(i, j) = 0.0f;
742
743 // Gyro rows 3-5: identity on angular-rate columns
744 C(3,0) = 1.0f;
745 C(4,1) = 1.0f;
746 C(5,2) = 1.0f;
747
748 return C;
749}

751// ─── Measurement prediction ───────────────────────────────────────────────────
752// Returns [accel_g (3), gyro_rad_s (3)]
753MeasVec Physics::predictMeasurement(float t, const StateVec& x, float u,
754 float alt) const {
755 const float w1=x(0,0), w2=x(1,0), w3=x(2,0);
756 const float v1=x(3,0), v2=x(4,0), v3=x(5,0);
757 const float qw=x(6,0), qx=x(7,0), qy=x(8,0), qz=x(9,0);
758
759 const WeatherSample wx = weatherAtAltitude(alt);
760 StateVec xdot = eom(t, x, u, alt);
761
762 // vdot (body) = xdot[3:6]
763 // Specific force = vdot + omega×v - g_body, divided by g to get g units
764 // omega × v
765 float oxv[3] = {
766 w2*v3 - w3*v2,
767 w3*v1 - w1*v3,
768 w1*v2 - w2*v1
769 };
770
771 // g_body = R_BW * [0,0,-g]
772 float R[3][3];
773 Physics::R_BW(qw, qx, qy, qz, R);
774 float g_body[3] = {
775 R[0][2] * (-wx.g),
776 R[1][2] * (-wx.g),
777 R[2][2] * (-wx.g)
778 };
779
780 MeasVec y;
781 for (int i = 0; i < 3; i++)
782 y(i, 0) = (xdot(3+i, 0) + oxv[i] - g_body[i]) / cfg.g_0;
783
784 // Gyro: direct angular rate observation
785 y(3,0) = w1;
786 y(4,0) = w2;
787 y(5,0) = w3;
788
789 return y;
790}
