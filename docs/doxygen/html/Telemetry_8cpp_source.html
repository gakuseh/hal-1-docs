67// ---------------------------------------------------------------------------
68// State machine:
69//
70// FC_MASTER (default) — FC transmits TX_BURST_SIZE telemetry packets, then
71// opens a short receive window. If a valid GND packet is decoded the
72// state switches to GND_MASTER.
73//
74// GND_MASTER — FC only receives; GND is the active transmitter. Each
75// successful decode resets the 5-second deadline. If the deadline
76// expires with no valid packet, FC reverts to FC_MASTER.
77//
78// Return value:
79// 0 — valid GND packet decoded this cycle
80// E22_NO_DATA — FC-master TX cycle; no receive attempted
81// E22_RECEIVE_ERR / E22_BAD_LENGTH / positive framing codes
82// — receive attempted but failed
83// --------------------------------------------------------------------------
84uint8_t Telemetry::Update()
85{
86 // Every 5 s, open a 2-second receive window — poll AUX continuously so
87 // any incoming GND packet is caught. If the window expires with no packet,
88 // fall through to the normal TX path.
89 static TickType_t lastWindowTick = 0u;
90 const TickType_t nowTick = xTaskGetTickCount();
91 if (nowTick - lastWindowTick >= pdMS_TO_TICKS(5000u)) {
92 lastWindowTick = nowTick;
93 const TickType_t deadline = nowTick + pdMS_TO_TICKS(2000u);
94 while (xTaskGetTickCount() < deadline) {
95 // If AUX is low the E22 is clocking an incoming packet onto the UART line.
96 // Receive it before transmitting so we don't overwrite the UART buffer.
97 if (!e22_aux_high()) {
98 uint8_t rx_status = receiveCommands(GNDOutData);
99 lastRxGood = (rx_status == 0);
100 if (rx_status == 0)
101 return 0;
102 }
103 }
104 }
105
106 // AUX high — module idle, transmit telemetry normally.
107 lastRxGood = false;
108 sendData(HALOutData);
109 return E22_NO_DATA;
110}

112// ---------------------------------------------------------------------------
113// Encode + transmit — FC sends telemetryData, GND receives it.
114// Frame layout (matching GND's decodeData exactly):
115// [0] GND_RADIO_ADDRHIGH \ fixed-point routing header —
116// [1] GND_RADIO_ADDRLOW > stripped by E22 module before delivery,
117// [2] CH915 / not seen by the remote MCU's UART
118// [3] SYNC1 (0xAA) \ start sync pair
119// [4] SYNC2 (0x55) /
120// [5] payload_len
121// [6] seq lo
122// [7] seq hi
123// [8 .. 8+N-1] payload (N bytes)
124// [8+N] CRC lo
125// [9+N] CRC hi
126// [10+N] SYNC2 (0x55) \ end sync pair — SYNC2 first,
127// [11+N] SYNC1 (0xAA) / SYNC1 is the final byte of the packet
128// ---------------------------------------------------------------------------
129template<typename T>
130uint8_t Telemetry::encodeAndSend(const T &payload)
131{
132 static_assert(sizeof(T) <= TELEMETRY_MAX_PAYLOAD - 12, "payload exceeds TX buffer size");
133 uint8_t payload_len = sizeof(T);
134
135 tx_buffer[0] = GND_RADIO_ADDRHIGH;
136 tx_buffer[1] = GND_RADIO_ADDRLOW;
137 tx_buffer[2] = GLOBAL_RADIO_CHANNEL;
138
139 // What is actually transmitted
140 tx_buffer[3] = TELEMETRY_SYNC1;
141 tx_buffer[4] = TELEMETRY_SYNC2;
142 tx_buffer[5] = payload_len;
143 tx_buffer[6] = lastSeq & 0xFF;
144 tx_buffer[7] = (lastSeq >> 8) & 0xFF;
145
146 memcpy(&tx_buffer[8], &payload, payload_len);
147
148 uint16_t crc = Checksum(&tx_buffer[3], 5 + payload_len); // covers SYNC1 → end of payload
149 tx_buffer[8 + payload_len] = crc & 0xFF;
150 tx_buffer[9 + payload_len] = (crc >> 8) & 0xFF;
151
152 // Ending sync pair: SYNC2 then SYNC1 (SYNC1 is the final byte)
153 tx_buffer[10 + payload_len] = TELEMETRY_SYNC2;
154 tx_buffer[11 + payload_len] = TELEMETRY_SYNC1;
155
156 int8_t status = transmit_e22_900t22s(tx_buffer, 12 + payload_len);
157 if(status != E22_OK)
158 return status;
159
160 lastSeq++;
161 return 0;
162}

164// ---------------------------------------------------------------------------
165// Decode a received frame from rx_buffer into payload.
166// buf_len is the number of bytes actually returned by recieve_e22_900t22s(),
167// bounding the sync search to valid data so stale buffer bytes cannot match.
168// Matches GND's Radio::decodeData exactly — all checks enabled.
169//
170// Expected frame layout (offsets relative to sync_idx):
171// [+0] SYNC1 (0xAA) \ start sync pair
172// [+1] SYNC2 (0x55) /
173// [+2] payload_len
174// [+3] seq lo
175// [+4] seq hi
176// [+5 .. +4+N] payload (N bytes)
177// [+5+N] CRC lo
178// [+6+N] CRC hi
179// [+7+N] SYNC2 (0x55) \ end sync pair — SYNC2 first,
180// [+8+N] SYNC1 (0xAA) / SYNC1 is the final byte of the packet
181// ---------------------------------------------------------------------------
182template<typename T>
183int8_t Telemetry::decodeData(T &payload, uint16_t buf_len)
184{
185 if (buf_len < 7u) return -1; // too short to contain any valid frame
186 // buf_len >= 7 here, so buf_len - 1u >= 6u — no underflow risk.
187
188 int16_t sync_idx = -1;
189 for(int16_t i = 0; i < buf_len; i++)
190 {
191 if(rx_buffer[i] == TELEMETRY_SYNC1 && rx_buffer[i + 1] == TELEMETRY_SYNC2)
192 {
193 sync_idx = i;
194 break;
195 }
196 }
197
198 if(sync_idx == -1)
199 return -1; // sync bytes not found in received data
200
201 // Enough bytes after sync for the full 5-byte header?
202 // [SYNC1][SYNC2][len][seq_lo][seq_hi] = 5 bytes before payload starts.
203 if((uint16_t)(sync_idx + 5) >= buf_len)
204 return -2;
205
206 uint8_t payload_len = rx_buffer[sync_idx + 2];
207
208 if(payload_len != sizeof(T))
209 return -3; // length mismatch — wrong packet type or struct size skew
210
211 // Full packet (header + payload + CRC + trailing sync pair) must fit inside received bytes.
212 // Layout after sync_idx: [S1][S2][len][seq_lo][seq_hi][payload x N][CRC_lo][CRC_hi][S2][S1]
213 // 0 1 2 3 4 5..4+N 5+N 6+N 7+N 8+N
214 if((uint16_t)(sync_idx + 5 + payload_len + 4) > buf_len)
215 return -4;
216
217 uint16_t seq_rx = rx_buffer[sync_idx + 3] | (rx_buffer[sync_idx + 4] << 8);
218
219 uint16_t crc_rx =
220 rx_buffer[sync_idx + 5 + payload_len] |
221 (rx_buffer[sync_idx + 6 + payload_len] << 8);
222
223 uint16_t crc_calc = Checksum(&rx_buffer[sync_idx], 5 + payload_len);
224
225 if(crc_rx != crc_calc)
226 return -5;
227
228 // Verify trailing sync bytes (mirror of header: SYNC2 then SYNC1)
229 if(rx_buffer[sync_idx + 7 + payload_len] != TELEMETRY_SYNC2 ||
230 rx_buffer[sync_idx + 8 + payload_len] != TELEMETRY_SYNC1)
231 return -6; // trailing sync mismatch — frame boundary corrupted
232
233 memcpy(&payload, &rx_buffer[sync_idx + 5], sizeof(T));
234
235 lastSeq = seq_rx;
236 return 0;
237}

248uint8_t Telemetry::receiveCommands(GndStationData &gnd)
249{
250 // Pass the full rx_buffer size so the entire framed packet is read.
251 // The framing overhead is 10 bytes (3-byte E22 header + 5-byte frame header
252 // + 2-byte CRC), so passing sizeof(GndStationData) would truncate the packet
253 // and make CRC verification impossible.
254
255 // reset rx_buffer to all zeroes in anticipation of new byte
256 memset(rx_buffer, 0, sizeof(rx_buffer));
257 vTaskSuspendAll();
258 int16_t len = recieve_e22_900t22s(rx_buffer, sizeof(GndStationData));
259 xTaskResumeAll();
260 if(len <= 0) return E22_RECEIVE_ERR;
261 if(len < 7) return E22_BAD_LENGTH;
262
263 int8_t status = decodeData(gnd, (uint16_t)len);
264 if(status != 0) return static_cast<uint8_t>(-status); // positive non-zero error code
265
266 // Propagate servo offsets to g_gndData so the CTRLs task can apply them.
267 // The CTRLs task reads g_gndData under g_ctrls_sensor_mutex; acquire it here
268 // too so the two-float update is atomic from its perspective.
269 if (osMutexAcquire(g_ctrls_sensor_mutex, 5) == osOK) {
270 // Idk why but for RCI to display these properly i have to switch them
271 // im tired boss
272 g_gndData.servoOffset1 = gnd.servoOffset2;
273 g_gndData.servoOffset2 = gnd.servoOffset1;
274 osMutexRelease(g_ctrls_sensor_mutex);
275 }
276
277 processCmd(gnd.CommandByte);
278 return 0;
279}

281// ---------------------------------------------------------------------------
282// Command handler — byte values must match GND's RADIO_DEFNS.h exactly.
283//
284// Default behaviour: echo the received command byte back in CommandResponseByte
285// so GND always gets positive confirmation that HAL received its instruction.
286// CommandResponseByte is only written here; no other path touches it, so the
287// echo-then-override pattern below is the single source of truth.
288//
289// Exception — HANDSHAKE (0xA1): GND expects the dedicated ACK byte 0xB2, not
290// a mirror of its own request byte, so that one case overrides the default.
291// ---------------------------------------------------------------------------
292void Telemetry::processCmd(uint8_t cmd)
293{
294 // Echo by default — GND inspects this field to confirm delivery.
295 HALOutData.CommandResponseByte = cmd;
296 if (cmd == SHUTDOWN_KEEPALIVE) {
297 // 217 — BYTE_ABORT
298 shutdownFlag = true;
299 g_ctrls_enabled = false; // zero servo outputs immediately
300 }
301 else {
302 // cmd is any number, a value of E22 frequncies. values 52-78.
303 if (des_cfg.REG2 != static_cast<R2_E22Channel915>(cmd)) changeOpFreq_e22_900t22s(static_cast<R2_E22Channel915>(cmd));
304 }
305}

307void Telemetry::processPyros(uint32_t pyroActivation)
308{
309 switch (pyroActivation) {
310 case PYROMAIN:
311 // Activate PID roll-damper indefinitely.
312 STATE_BACKUP_PID = true;
313 ALLOW_ACTUATION = true;
314 g_ctrls_enabled = true;
315 break;
316
317 case PYRODROGUEBKP:
318 // Activate state-space controller from t = 0.
319 g_ctrls_reset_time = true; // CTRLs_STATESPACE resets launch_start_ms on next tick
320 STATE_BACKUP_PID = false;
321 ALLOW_ACTUATION = true;
322 g_ctrls_enabled = true;
323 break;
324
325 case PYRODROGUEMAIN:
326 // Kill both algorithms.
327 g_ctrls_enabled = false;
328 ALLOW_ACTUATION = false;
329 break;
330
331 default:
332 break;
333 }
