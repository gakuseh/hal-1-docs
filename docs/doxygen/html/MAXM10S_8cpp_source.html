22uint16_t MAXM10S::getAvailableBytes()
23{
24 uint8_t buf[2] = {0};
25 // 2-byte register read at 400kHz I2C takes <1ms; 20ms is a generous ceiling
26 if (HAL_I2C_Mem_Read(_hi2c, I2C_ADDRESS, REG_BYTES_AVAIL,
27 I2C_MEMADD_SIZE_8BIT, buf, 2, 50) != HAL_OK)
28 return 0;
29 uint16_t n = (uint16_t)((buf[0] << 8) | buf[1]);
30 return (n == 0xFFFF) ? 0 : n; // 0xFFFF = I2C FIFO stall; treat as 0
31}

33uint16_t MAXM10S::readStream(uint16_t numBytes)
34{
35 if (numBytes == 0) return 0;
36 if (numBytes > RX_BUF_SIZE) numBytes = RX_BUF_SIZE;
37 // At 400kHz I2C, 1024 bytes takes ~23ms; 50ms is a safe ceiling
38 if (HAL_I2C_Mem_Read(_hi2c, I2C_ADDRESS, REG_DATA_STREAM,
39 I2C_MEMADD_SIZE_8BIT, rxBuf, numBytes, 100) != HAL_OK)
40 return 0;
41 return numBytes;
42}

48uint8_t MAXM10S::Init(I2C_HandleTypeDef *hi2c)
49{
50 _hi2c = hi2c;
51
52 // Release reset and wait for module boot (~300 ms minimum)
53 HAL_GPIO_WritePin(GPS_RST_GPIO_Port, GPS_RST_Pin, GPIO_PIN_SET);
54 osDelay(500);
55
56 // Verify module is reachable over I2C before sending config.
57 uint8_t probe[2] = {0xFF, 0xFF};
58 if (HAL_I2C_Mem_Read(_hi2c, I2C_ADDRESS, REG_BYTES_AVAIL,
59 I2C_MEMADD_SIZE_8BIT, probe, 2, 50) != HAL_OK)
60 return 1;
61
62 // UBX-CFG-VALSET: configure dynamic model and enable NMEA on DDC (I2C).
63 //
64 // NOTE: UBX-CFG-NAV5 (0x06/0x24) is NOT supported on the M10 platform —
65 // the module silently NAKs it and keeps its default "portable" model.
66 // M10 uses UBX-CFG-VALSET (0x06/0x8A) for all runtime configuration.
67 //
68 // Keys configured (all 1-byte values, stored little-endian):
69 // CFG-NAVSPG-DYNMODEL 0x20110021 = 8 (Airborne <4g; 7 = <2g)
70 // CFG-MSGOUT-NMEA_ID_GGA_I2C 0x209100BB = 1 (position, altitude, sats)
71 // CFG-MSGOUT-NMEA_ID_RMC_I2C 0x209100AC = 1 (lat/lon, speed, time)
72 // CFG-MSGOUT-NMEA_ID_GSA_I2C 0x209100C0 = 1 (HDOP, fix type)
73 //
74 // Payload = 4 (header) + 4×5 (key-value pairs) = 24 bytes.
75 // Fletcher-8 checksum over class+ID+length+payload: CK_A=0x46, CK_B=0x64.
76 static const uint8_t ubx_valset[] = {
77 0xB5, 0x62, // sync chars
78 0x06, 0x8A, // class=CFG, ID=VALSET
79 0x18, 0x00, // payload length = 24 (LE)
80 // payload ─────────────────────────────────────────────────────────
81 0x00, // version
82 0x07, // layers: RAM + BBR + Flash
83 0x00, 0x00, // reserved
84 0x21, 0x00, 0x11, 0x20, 0x08, // CFG-NAVSPG-DYNMODEL = 8
85 0xBB, 0x00, 0x91, 0x20, 0x01, // CFG-MSGOUT-NMEA_ID_GGA_I2C = 1
86 0xAC, 0x00, 0x91, 0x20, 0x01, // CFG-MSGOUT-NMEA_ID_RMC_I2C = 1
87 0xC0, 0x00, 0x91, 0x20, 0x01, // CFG-MSGOUT-NMEA_ID_GSA_I2C = 1
88 // checksum ────────────────────────────────────────────────────────
89 0x46, 0x64 // CK_A, CK_B
90 };
91 sendUBX(ubx_valset, sizeof(ubx_valset));
92
93 return 0;
94}

96uint8_t MAXM10S::update()
97{
98 uint16_t available = getAvailableBytes();
99 if (available == 0) return GPS_NO_BYTES;
100
101 uint16_t bytesRead = readStream(available);
102 if (bytesRead == 0) return GPS_READ_FAIL;
103
104 bool newData = false;
105
106 for (uint16_t i = 0; i < bytesRead; i++) {
107 uint8_t b = rxBuf[i];
108 if (b == 0xFF) continue; // u-blox DDC filler byte — no data pending
109
110 if (!gpsParser.encode(b)) continue; // mid-sentence — keep feeding
111
112 // A complete NMEA sentence was parsed. Update all fields unconditionally
113 // so callers always see current satellite/quality state even without a fix.
114
115 if (gpsParser.satellites.isValid())
116 gpsDataStruct.satellites = (uint8_t)gpsParser.satellites.value();
117
118 if (gpsParser.hdop.isValid())
119 gpsDataStruct.hdop = (float)gpsParser.hdop.hdop();
120
121 gpsDataStruct.isLocated = gpsParser.location.isValid();
122 gpsDataStruct.isAltituded = gpsParser.altitude.isValid();
123 gpsDataStruct.isTimed = gpsParser.time.isValid();
124 gpsDataStruct.hasFix = gpsDataStruct.isLocated && gpsDataStruct.isAltituded;
125
126 if (gpsParser.location.isUpdated() && gpsDataStruct.isLocated) {
127 gpsDataStruct.latitude = gpsParser.location.lat();
128 gpsDataStruct.longitude = gpsParser.location.lng();
129 gpsDataStruct.lastFixMs = HAL_GetTick();
130 newData = true;
131 }
132
133 if (gpsParser.altitude.isUpdated() && gpsDataStruct.isAltituded) {
134 gpsDataStruct.altitude = gpsParser.altitude.meters();
135 newData = true;
136 }
137
138 if (gpsParser.speed.isUpdated() && gpsParser.speed.isValid()) {
139 gpsDataStruct.speedKmh = (float)gpsParser.speed.kmph();
140 newData = true;
141 }
142
143 if (gpsParser.time.isUpdated() && gpsDataStruct.isTimed) {
144 gpsDataStruct.hour = gpsParser.time.hour();
145 gpsDataStruct.minute = gpsParser.time.minute();
146 gpsDataStruct.second = gpsParser.time.second();
147 newData = true;
148 }
149 }
150
151 return newData ? GPS_NEW_DATA : GPS_NO_NEW_SENTENCE;
152}
