43float ControlLaw::gainSchedule_(float t, const StateVec& xhat, float alt) const {
44 // Air-relative speed at the model/weather altitude, matching the EOM.
45 const WeatherSample wx = phys_.weatherAtAltitude(alt);
46 float va[3];
47 phys_.airRelativeVelocity(xhat, wx.wind_x, wx.wind_y, va);
48 const float vmag = sqrtf(va[0]*va[0] + va[1]*va[1] + va[2]*va[2]);
49 if (vmag < cfg_.min_control_speed) return 0.0f;
50
51 const float dM_dzeta = phys_.canardMomentJacobian(vmag); // N·m/rad
52 if (fabsf(dM_dzeta) < 1e-12f) return 0.0f;
53
54 const float I3 = phys_.getInertiaRoll(t);
55 return I3 * cfg_.roll_damping_lambda / dM_dzeta;
56}

58float ControlLaw::computeControl(float t, const StateVec& xhat, float alt) {
59 const float dt = t - last_t_;
60 last_t_ = t;
61
62 // Roll rate error (reference = 0)
63 const float error = xhat(2, 0); // w3
64
65 // Gain-scheduled proportional output (sign incorporates roll effectiveness)
66 const float K = gainSchedule_(t, xhat, alt);
67 float u_cmd = -K * rem_sign_ * error;
68
69 // Rate limit
70 if (dt > 0.0f && dt < 0.5f) {
71 const float du_max = cfg_.max_deflection_rate * dt;
72 if (u_cmd - last_u_ > du_max) u_cmd = last_u_ + du_max;
73 if (u_cmd - last_u_ < -du_max) u_cmd = last_u_ - du_max;
74 }
75
76 // Saturation
77 if (u_cmd > cfg_.max_deflection_rad) u_cmd = cfg_.max_deflection_rad;
78 if (u_cmd < -cfg_.max_deflection_rad) u_cmd = -cfg_.max_deflection_rad;
79
80 // IREC compliance: inhibit during motor burn
81 if (cfg_.irec_compliant && t < cfg_.t_burnout) u_cmd = 0.0f;
82
83 // Pre-launch: no command
84 if (t <= 0.0f) u_cmd = 0.0f;
85
86 last_u_ = u_cmd;
87 return u_cmd;
88}

90// ─── Roll effectiveness monitor ───────────────────────────────────────────────
91// Call once per loop with the CURRENT gyro w3, the PREVIOUS command in rad,
92// and the current EKF speed estimate. Flips rem_sign_ when the measured roll
93// acceleration consistently opposes the modeled canard moment.
94void ControlLaw::updateRollEffectivenessSign(float t, float w3_meas,
95 float u_prev_rad, float vmag) {
96 if (!cfg_.rem_enabled) return;
97
98 const float prev_t = rem_prev_t_;
99 const float prev_w3 = rem_prev_w3_;
100 rem_prev_t_ = t;
101 rem_prev_w3_ = w3_meas;
102
103 if (prev_t < 0.0f) return; // first call — no previous sample yet
104
105 const float dt = t - prev_t;
106 if (dt <= 0.0f) { rem_mismatch_count_ = 0; return; }
107
108 // Inhibit during burn and post-burn settling period
109 if (t < cfg_.t_burnout + cfg_.rem_post_burn_delay_s) {
110 rem_mismatch_count_ = 0; return;
111 }
112
113 // Skip if command is too small to produce a reliable signal
114 if (fabsf(u_prev_rad) < cfg_.rem_min_cmd_rad) {
115 rem_mismatch_count_ = 0; return;
116 }
117
118 // Expected roll acceleration from the canard moment model
119 const float I3 = phys_.getInertiaRoll(t);
120 const float M_cfd = phys_.canardMoment(vmag, u_prev_rad);
121 const float expected_accel = (I3 > 1e-10f) ? (M_cfd / I3) : 0.0f;
122
123 // Measured roll acceleration from consecutive gyro readings
124 const float measured_accel = (w3_meas - prev_w3) / dt;
125
126 // Require both sides to be significant before trusting the comparison
127 if (fabsf(expected_accel) < cfg_.rem_min_expected_accel ||
128 fabsf(measured_accel) < cfg_.rem_min_measured_accel) {
129 rem_mismatch_count_ = 0; return;
130 }
131
132 // Accumulate or reset mismatch counter
133 if (measured_accel * expected_accel < 0.0f) {
134 rem_mismatch_count_++;
135 } else {
136 rem_mismatch_count_ = 0;
137 }
138
139 // Flip sign once threshold is reached
140 const bool can_flip = cfg_.rem_allow_flip_back || (rem_flip_count_ == 0);
141 if (can_flip && rem_mismatch_count_ >= cfg_.rem_required_mismatches) {
142 rem_sign_ *= -1.0f;
143 rem_flip_count_++;
144 rem_mismatch_count_ = 0;
145 }
146}
